1//! Build registered Serenade SPA apps with Trunk.
2
3use clap::Parser;
4use serenade_cli::mode::{discover_apps, find_examples_root, select_apps};
5
6#[derive(Debug, Parser)]
7#[command(
8    name = "serenade-spa",
9    about = "Build a registered Serenade SPA app with Trunk"
10)]
11struct Cli {
12    /// Build every app registered with mode = "spa".
13    #[arg(long, conflicts_with = "app")]
14    all: bool,
15    /// Build the app named APP (defaults to demo-spa).
16    app: Option<String>,
17}
18
19fn main() -> anyhow::Result<()> {
20    let cli = Cli::parse();
21    let root = find_examples_root(&std::env::current_dir()?)?;
22    let apps = discover_apps(&root.join("examples"))?;
23    let selected = select_apps(&apps, "spa", "demo-spa", cli.all, cli.app.as_deref())?;
24
25    for app in selected {
26        println!("Building {} ({})", app.name, app.directory.display());
27        serenade_cli::trunk::build(&app.directory, &[]).map_err(|error| {
28            anyhow::anyhow!("failed to build SPA app `{}`: {error:#}", app.name)
29        })?;
30        let dist = serenade_cli::html_output::dist_directory_for_build(&app.directory, &[])?;
31        let minified = serenade_cli::html_output::minify_html_files(&dist)?;
32        println!("Minified {minified} HTML document(s) for {}", app.name);
33    }
34
35    Ok(())
36}
