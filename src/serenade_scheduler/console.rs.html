10/// Runs the default scheduler loop (or a single tick with `--once`).
11#[derive(Clone, Copy, Debug, Default)]
12pub struct RunSchedulerCommand;
13
14impl Command for RunSchedulerCommand {
15    fn name(&self) -> &'static str {
16        "serenade:scheduler:run"
17    }
18
19    fn description(&self) -> &'static str {
20        "Run the scheduler tick loop (use --once for a single tick)"
21    }
22
23    fn execute(&self, input: &Input) -> Result<(), ConsoleError> {
24        let Some(container) = input.container() else {
25            return Err(ConsoleError::Failed(
26                "serenade:scheduler:run needs a container (Application::run_with)".to_owned(),
27            ));
28        };
29        let service = container
30            .get_as::<SchedulerService>(DEFAULT_SCHEDULER_SERVICE)
31            .map_err(|error| ConsoleError::Failed(error.to_string()))?;
32        let once = input.args().iter().any(|arg| arg == "--once");
33        run_scheduler_loop(&service, &SystemClock, once)
34    }
35}

37#[derive(Debug, PartialEq, Eq)]
38pub enum LoopStep {
39    /// Sleep then tick again.
40    Continue {
41        /// How long to sleep before the next tick.
42        sleep: Duration,
43    },
44    /// Stop the loop.
45    Done,
46}
47
48/// One tick of the scheduler run loop (no sleep). Used by [`run_scheduler_loop`].
49///
50/// # Errors
51///
52/// Returns [`ConsoleError::Failed`] when dispatch fails.
53pub fn scheduler_loop_step(
54    service: &SchedulerService,
55    clock: &dyn Clock,
56    once: bool,
57) -> Result<LoopStep, ConsoleError> {
58    let due = service
59        .tick_and_dispatch(clock)
60        .map_err(|error| ConsoleError::Failed(error.to_string()))?;
61    for job in &due {
62        println!(
63            "scheduler: dispatched `{}` (scheduled_for={:?})",
64            job.id(),
65            job.scheduled_for()
66        );
67    }
68    if once {
69        return Ok(LoopStep::Done);
70    }
71    if service.is_empty() {
72        println!("scheduler: no schedules registered; exiting");
73        return Ok(LoopStep::Done);
74    }
75    let wake = service
76        .next_wake()
77        .expect("non-empty scheduler always has a next wake");
78    let now = clock.now();
79    let sleep = wake
80        .duration_since(now)
81        .unwrap_or(Duration::ZERO)
82        .min(Duration::from_secs(60));
83    Ok(LoopStep::Continue { sleep })
84}

86/// Tick / dispatch loop used by [`RunSchedulerCommand`] (testable with [`crate::ManualClock`]).
87///
88/// # Errors
89///
90/// Returns [`ConsoleError::Failed`] when dispatch fails.
91pub fn run_scheduler_loop(
92    service: &SchedulerService,
93    clock: &dyn Clock,
94    once: bool,
95) -> Result<(), ConsoleError> {
96    run_scheduler_loop_limited(service, clock, once, None)
97}
98
99/// Like [`run_scheduler_loop`], but stops after `max_steps` iterations when set (tests).
100///
101/// # Errors
102///
103/// Returns [`ConsoleError::Failed`] when dispatch fails.
104pub fn run_scheduler_loop_limited(
105    service: &SchedulerService,
106    clock: &dyn Clock,
107    once: bool,
108    max_steps: Option<usize>,
109) -> Result<(), ConsoleError> {
110    let limit = max_steps.unwrap_or(usize::MAX);
111    for _ in 0..limit {
112        match scheduler_loop_step(service, clock, once)? {
113            LoopStep::Done => return Ok(()),
114            LoopStep::Continue { sleep } => thread::sleep(sleep),
115        }
116    }
117    Ok(())
118}
