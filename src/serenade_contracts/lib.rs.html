28/// Marker for entity identifiers passed into repository traits.
29pub trait EntityId: Clone + PartialEq + Eq + Send + Sync + std::fmt::Debug + 'static {
30    /// String form used in errors and logs.
31    fn as_str(&self) -> &str;
32}
33
34impl EntityId for String {
35    fn as_str(&self) -> &str {
36        self.as_str()
37    }
38}
