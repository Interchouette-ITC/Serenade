3use crate::{EntityId, RepositoryError};
4use std::future::Future;
5
6/// Order persistence for checkout and fulfillment hooks.
7///
8/// # Examples
9///
10/// ```
11/// use serenade_contracts::{OrderRepository, PersistenceError};
12/// use std::future::Future;
13///
14/// struct MemoryOrders;
15///
16/// impl OrderRepository for MemoryOrders {
17///     type Error = PersistenceError;
18///     type Id = String;
19///     type Order = String;
20///
21///     fn find_by_number(
22///         &self,
23///         _number: &str,
24///     ) -> impl Future<Output = Result<Option<Self::Order>, Self::Error>> + Send {
25///         async move { Ok(None) }
26///     }
27///
28///     fn save(
29///         &self,
30///         _order: &Self::Order,
31///     ) -> impl Future<Output = Result<(), Self::Error>> + Send {
32///         async move { Ok(()) }
33///     }
34///
35///     fn save_idempotent(
36///         &self,
37///         _order: &Self::Order,
38///         _idempotency_key: &str,
39///     ) -> impl Future<Output = Result<(), Self::Error>> + Send {
40///         async move { Ok(()) }
41///     }
42/// }
43/// ```
44pub trait OrderRepository: Send + Sync {
45    /// Error type for this adapter.
46    type Error: RepositoryError;
47    /// Order identifier type.
48    type Id: EntityId;
49    /// Application-defined order aggregate.
50    type Order: Send + Sync;
51
52    /// Load an order by human-readable number.
53    fn find_by_number(
54        &self,
55        number: &str,
56    ) -> impl Future<Output = Result<Option<Self::Order>, Self::Error>> + Send;
57
58    /// Insert or update an order aggregate.
59    fn save(&self, order: &Self::Order) -> impl Future<Output = Result<(), Self::Error>> + Send;
60
61    /// Idempotent checkout insert keyed by client idempotency token.
62    fn save_idempotent(
63        &self,
64        order: &Self::Order,
65        idempotency_key: &str,
66    ) -> impl Future<Output = Result<(), Self::Error>> + Send;
67}
