1//! Category read repository contract.
2
3use crate::{EntityId, PageRequest, RepositoryError};
4use std::future::Future;
5
6/// Category tree reads for navigation and catalog filters.
7///
8/// # Examples
9///
10/// ```
11/// use serenade_contracts::{CategoryRepository, PageRequest, PersistenceError};
12/// use std::future::Future;
13///
14/// struct MemoryCategories;
15///
16/// impl CategoryRepository for MemoryCategories {
17///     type Error = PersistenceError;
18///     type Id = String;
19///     type Category = String;
20///
21///     fn find_by_id(
22///         &self,
23///         _id: &Self::Id,
24///     ) -> impl Future<Output = Result<Option<Self::Category>, Self::Error>> + Send {
25///         async move { Ok(None) }
26///     }
27///
28///     fn find_by_slug(
29///         &self,
30///         _slug: &str,
31///         _parent_id: Option<&Self::Id>,
32///     ) -> impl Future<Output = Result<Option<Self::Category>, Self::Error>> + Send {
33///         async move { Ok(None) }
34///     }
35///
36///     fn list_children(
37///         &self,
38///         _parent_id: Option<&Self::Id>,
39///         _page: PageRequest,
40///     ) -> impl Future<Output = Result<Vec<Self::Category>, Self::Error>> + Send {
41///         async move { Ok(Vec::new()) }
42///     }
43/// }
44/// ```
45pub trait CategoryRepository: Send + Sync {
46    /// Error type for this adapter.
47    type Error: RepositoryError;
48    /// Category identifier type.
49    type Id: EntityId;
50    /// Application-defined category model.
51    type Category: Send + Sync;
52
53    /// Load one category by primary key.
54    fn find_by_id(
55        &self,
56        id: &Self::Id,
57    ) -> impl Future<Output = Result<Option<Self::Category>, Self::Error>> + Send;
58
59    /// Load one category by slug within an optional parent scope.
60    fn find_by_slug(
61        &self,
62        slug: &str,
63        parent_id: Option<&Self::Id>,
64    ) -> impl Future<Output = Result<Option<Self::Category>, Self::Error>> + Send;
65
66    /// List child categories for a parent (`None` = roots).
67    fn list_children(
68        &self,
69        parent_id: Option<&Self::Id>,
70        page: PageRequest,
71    ) -> impl Future<Output = Result<Vec<Self::Category>, Self::Error>> + Send;
72}
