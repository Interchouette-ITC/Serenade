1//! Incoming HTTP request.
2
3use crate::{AttributeBag, Headers, Method};
4
5/// Framework-agnostic HTTP request.
6///
7/// Adapters copy method, path, headers, and body from the server crate.
8#[derive(Debug)]
9pub struct Request {
10    method: Method,
11    path: String,
12    headers: Headers,
13    body: Vec<u8>,
14    attributes: AttributeBag,
15}
16
17impl Request {
18    /// Builds a request with an empty body, headers, and attributes.
19    #[must_use]
20    pub fn new(method: Method, path: impl Into<String>) -> Self {
21        Self {
22            method,
23            path: path.into(),
24            headers: Headers::new(),
25            body: Vec::new(),
26            attributes: AttributeBag::new(),
27        }
28    }
29
30    /// Replaces the header map after insert.
31    #[must_use]
32    pub fn with_header(mut self, name: impl AsRef<str>, value: impl Into<String>) -> Self {
33        self.headers.insert(name, value);
34        self
35    }
36
37    /// Sets the raw body.
38    #[must_use]
39    pub fn with_body(mut self, body: impl Into<Vec<u8>>) -> Self {
40        self.body = body.into();
41        self
42    }
43
44    /// HTTP method.
45    #[must_use]
46    pub const fn method(&self) -> Method {
47        self.method
48    }
49
50    /// Path (no query string). Query parsing is left to routing or the adapter.
51    #[must_use]
52    pub fn path(&self) -> &str {
53        &self.path
54    }
55
56    /// Request headers.
57    #[must_use]
58    pub const fn headers(&self) -> &Headers {
59        &self.headers
60    }
61
62    /// Mutable headers (adapters and middleware).
63    pub fn headers_mut(&mut self) -> &mut Headers {
64        &mut self.headers
65    }
66
67    /// Raw body bytes.
68    #[must_use]
69    pub fn body(&self) -> &[u8] {
70        &self.body
71    }
72
73    /// Request attributes.
74    #[must_use]
75    pub const fn attributes(&self) -> &AttributeBag {
76        &self.attributes
77    }
78
79    /// Mutable attributes (middleware, router, controller).
80    pub fn attributes_mut(&mut self) -> &mut AttributeBag {
81        &mut self.attributes
82    }
83}
