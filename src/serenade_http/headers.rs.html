1//! Case-insensitive HTTP header map (last value wins).
2
3use std::collections::HashMap;
4
5/// Header names stored in lowercase; lookup is case-insensitive.
6#[derive(Clone, Debug, Default, Eq, PartialEq)]
7pub struct Headers {
8    values: HashMap<String, String>,
9}
10
11impl Headers {
12    /// Empty header map.
13    #[must_use]
14    pub fn new() -> Self {
15        Self::default()
16    }
17
18    /// Inserts or replaces a header. The name is stored in lowercase.
19    pub fn insert(&mut self, name: impl AsRef<str>, value: impl Into<String>) {
20        self.values
21            .insert(name.as_ref().to_ascii_lowercase(), value.into());
22    }
23
24    /// Returns the value for `name`, if present.
25    #[must_use]
26    pub fn get(&self, name: &str) -> Option<&str> {
27        self.values
28            .get(&name.to_ascii_lowercase())
29            .map(String::as_str)
30    }
31
32    /// Number of stored headers.
33    #[must_use]
34    pub fn len(&self) -> usize {
35        self.values.len()
36    }
37
38    /// Whether the map is empty.
39    #[must_use]
40    pub fn is_empty(&self) -> bool {
41        self.values.is_empty()
42    }
43
44    /// Iterates `(lowercase name, value)` pairs.
45    pub fn iter(&self) -> impl Iterator<Item = (&str, &str)> {
46        self.values.iter().map(|(k, v)| (k.as_str(), v.as_str()))
47    }
48}
