1//! Symfony-shaped named locks.
2//!
3//! - [`LockStore`]: persist ownership for a resource + token
4//! - [`LockFactory`] / [`Lock`]: acquire, release, refresh, `is_acquired`
5//! - [`InMemoryLockStore`]: process-local store with optional TTL
6//! - [`FilesystemLockStore`]: disk-backed store
7//! - [`RedisLockStore`] (feature `redis`): Redis-backed store
8//! - [`RegisterDefaultLockPass`]: DI default service id [`DEFAULT_LOCK_STORE_SERVICE`]
9//!
10//! # Examples
11//!
12//! ```
13//! use std::sync::Arc;
14//! use std::time::Duration;
15//! use serenade_lock::{InMemoryLockStore, LockFactory};
16//!
17//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
18//! let store = Arc::new(InMemoryLockStore::new());
19//! let factory = LockFactory::new(store);
20//! let lock = factory.create_lock("cron:mail-blast", Some(Duration::from_secs(30)), true)?;
21//! assert!(lock.acquire()?);
22//! lock.release()?;
23//! # Ok(())
24//! # }
25//! ```
