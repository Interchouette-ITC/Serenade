16/// Runs an interactive recipe picker and applies the selection.
17///
18/// # Errors
19///
20/// Non-TTY, IO, apply failure, or user cancel (`q` / Esc).
21pub fn run_recipe_tui(options: &ApplyOptions) -> Result<Recipe, CliError> {
22    if !io::stdin().is_terminal() || !io::stdout().is_terminal() {
23        return Err(CliError::NotATty);
24    }
25
26    let recipes = list_recipes();
27    if recipes.is_empty() {
28        return Err(CliError::InvalidRecipe("no embedded recipes".into()));
29    }
30
31    let selection = pick_recipe(&recipes, options)?;
32    let Some(id) = selection else {
33        return Err(CliError::Cancelled);
34    };
35
36    let recipe = apply_recipe(&id, options)?;
37    println!(
38        "applied recipe `{}` into {}",
39        recipe.id,
40        options.root.display()
41    );
42    print_hints(&recipe);
43    Ok(recipe)
44}

46fn pick_recipe(
47    recipes: &[(String, String)],
48    options: &ApplyOptions,
49) -> Result<Option<String>, CliError> {
50    enable_raw_mode()?;
51    stdout().execute(EnterAlternateScreen)?;
52    let result = run_picker(recipes, options);
53    stdout().execute(LeaveAlternateScreen)?;
54    disable_raw_mode()?;
55    result
56}

58fn run_picker(
59    recipes: &[(String, String)],
60    options: &ApplyOptions,
61) -> Result<Option<String>, CliError> {
62    let mut terminal = Terminal::new(CrosstermBackend::new(stdout()))?;
63    let mut state = ListState::default().with_selected(Some(0));
64
65    loop {
66        terminal.draw(|frame| {
67            render(frame, recipes, options, &mut state);
68        })?;
69
70        if !event::poll(std::time::Duration::from_millis(200))? {
71            continue;
72        }
73        let Event::Key(key) = event::read()? else {
74            continue;
75        };
76        if key.kind != KeyEventKind::Press {
77            continue;
78        }
79        match key.code {
80            KeyCode::Char('q') | KeyCode::Esc => return Ok(None),
81            KeyCode::Enter => {
82                let index = state.selected().unwrap_or(0);
83                return Ok(recipes.get(index).map(|(id, _)| id.clone()));
84            }
85            KeyCode::Down | KeyCode::Char('j') => {
86                let next = state
87                    .selected()
88                    .map_or(0, |i| (i + 1).min(recipes.len() - 1));
89                state.select(Some(next));
90            }
91            KeyCode::Up | KeyCode::Char('k') => {
92                let next = state.selected().map_or(0, |i| i.saturating_sub(1));
93                state.select(Some(next));
94            }
95            _ => {}
96        }
97    }
98}

100fn render(
101    frame: &mut ratatui::Frame,
102    recipes: &[(String, String)],
103    options: &ApplyOptions,
104    state: &mut ListState,
105) {
106    let area = frame.area();
107    let chunks = Layout::vertical([
108        Constraint::Length(3),
109        Constraint::Min(5),
110        Constraint::Length(3),
111    ])
112    .split(area);
113
114    let header = Paragraph::new(format!(
115        "root={}  force={}  no-cargo={}  (↑/↓ move, Enter apply, q quit)",
116        options.root.display(),
117        options.force,
118        options.no_cargo
119    ))
120    .block(
121        Block::default()
122            .borders(Borders::ALL)
123            .title("serenade tui - pick a recipe"),
124    );
125    frame.render_widget(header, chunks[0]);
126
127    let items: Vec<ListItem> = recipes
128        .iter()
129        .map(|(id, description)| ListItem::new(format!("{id}  -  {description}")))
130        .collect();
131    let list = List::new(items)
132        .block(Block::default().borders(Borders::ALL).title("Recipes"))
133        .highlight_style(Style::new().bold().reversed())
134        .highlight_symbol("> ");
135    frame.render_stateful_widget(list, chunks[1], state);
136
137    let footer = Paragraph::new("Same apply path as `serenade recipe apply <id>`")
138        .block(Block::default().borders(Borders::ALL));
139    frame.render_widget(footer, chunks[2]);
140}
