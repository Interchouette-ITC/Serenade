3use std::path::Path;

5use anyhow::{Context, bail};
6use serde::Deserialize;
7use walkdir::WalkDir;

9#[derive(Deserialize)]
10struct TrunkConfig {
11    build: Option<TrunkBuild>,
12}
13
14#[derive(Deserialize)]
15struct TrunkBuild {
16    dist: Option<std::path::PathBuf>,
17}
18
19/// Resolves Trunk's configured output directory, defaulting to `dist/`.
20///
21/// # Errors
22///
23/// Returns an error when an existing `Trunk.toml` cannot be read or parsed.
24pub fn dist_directory(app: &Path) -> anyhow::Result<std::path::PathBuf> {
25    let config_path = app.join("Trunk.toml");
26    let config = match std::fs::read_to_string(&config_path) {
27        Ok(contents) => Some(
28            toml::from_str::<TrunkConfig>(&contents)
29                .with_context(|| format!("failed to parse {}", config_path.display()))?,
30        ),
31        Err(error) if error.kind() == std::io::ErrorKind::NotFound => None,
32        Err(error) => {
33            return Err(error).with_context(|| format!("failed to read {}", config_path.display()));
34        }
35    };
36    let dist = config
37        .and_then(|config| config.build)
38        .and_then(|build| build.dist)
39        .unwrap_or_else(|| std::path::PathBuf::from("dist"));
40    Ok(if dist.is_absolute() {
41        dist
42    } else {
43        app.join(dist)
44    })
45}

47/// Minifies every UTF-8 `.html` file below `dist`.
48///
49/// # Errors
50///
51/// Returns an error when the output directory is missing, scanning fails, a
52/// document cannot be read, or writing its minified form fails.
53pub fn minify_html_files(dist: &Path) -> anyhow::Result<usize> {
54    if !dist.is_dir() {
55        bail!("HTML output directory does not exist: {}", dist.display());
56    }
57
58    let mut count = 0;
59    for entry in WalkDir::new(dist) {
60        let entry = entry.with_context(|| format!("failed to scan {}", dist.display()))?;
61        if !entry.file_type().is_file()
62            || !entry
63                .path()
64                .extension()
65                .and_then(|extension| extension.to_str())
66                .is_some_and(|extension| extension.eq_ignore_ascii_case("html"))
67        {
68            continue;
69        }
70
71        let path = entry.path();
72        let html = std::fs::read_to_string(path)
73            .with_context(|| format!("failed to read HTML document {}", path.display()))?;
74        let minified = serenade_minify::minify_html(&html)
75            .with_context(|| format!("failed to minify HTML document {}", path.display()))?;
76        std::fs::write(path, minified).with_context(|| {
77            format!("failed to write minified HTML document {}", path.display())
78        })?;
79        count += 1;
80    }
81
82    if count == 0 {
83        bail!("no HTML documents found in {}", dist.display());
84    }
85    Ok(count)
86}

88#[cfg(test)]
89mod tests {
90    use super::*;
91
92    #[test]
93    fn minifies_every_html_document_and_leaves_other_assets_untouched() {
94        let temp = tempfile::tempdir().expect("temporary directory");
95        let nested = temp.path().join("about");
96        std::fs::create_dir(&nested).expect("nested directory");
97        let home = temp.path().join("index.html");
98        let about = nested.join("index.html");
99        let css = temp.path().join("site.css");
100        std::fs::write(
101            &home,
102            "<!doctype html>\n<html><body><p> hello </p></body></html>",
103        )
104        .expect("home HTML");
105        std::fs::write(&about, "<html>\n<body><p>world</p></body>\n</html>").expect("about HTML");
106        std::fs::write(&css, " body { color: red; }").expect("CSS asset");
107
108        assert_eq!(minify_html_files(temp.path()).expect("minify files"), 2);
109        assert_eq!(
110            std::fs::read_to_string(home).expect("read home"),
111            "<!doctype html><html><body><p>hello</p></body></html>"
112        );
113        assert_eq!(
114            std::fs::read_to_string(about).expect("read about"),
115            "<html><body><p>world</p></body></html>"
116        );
117        assert_eq!(
118            std::fs::read_to_string(css).expect("read CSS"),
119            " body { color: red; }"
120        );
121    }
122
123    #[test]
124    fn rejects_missing_or_empty_output_directories() {
125        let temp = tempfile::tempdir().expect("temporary directory");
126        assert!(minify_html_files(&temp.path().join("missing")).is_err());
127        assert!(minify_html_files(temp.path()).is_err());
128    }
129
130    #[test]
131    fn resolves_trunk_dist_with_default_and_custom_paths() {
132        let temp = tempfile::tempdir().expect("temporary directory");
133        assert_eq!(
134            dist_directory(temp.path()).expect("default dist"),
135            temp.path().join("dist")
136        );
137
138        std::fs::write(
139            temp.path().join("Trunk.toml"),
140            "[build]\ndist = \"public/site\"\n",
141        )
142        .expect("Trunk config");
143        assert_eq!(
144            dist_directory(temp.path()).expect("configured dist"),
145            temp.path().join("public/site")
146        );
147    }
