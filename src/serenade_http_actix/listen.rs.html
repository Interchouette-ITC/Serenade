1//! Bind and run Actix Web with a Serenade [`HttpKernel`].
2
3use std::net::ToSocketAddrs;
4
5use actix_web::body::MessageBody;
6use actix_web::dev::{ServiceFactory, ServiceRequest, ServiceResponse};
7use actix_web::{web, App, Error, HttpRequest, HttpResponse, HttpServer};
8use serenade_http::HttpKernel;
9
10use crate::dispatch;
11
12/// Builds the Actix app that forwards every request to `kernel` via [`dispatch`].
13#[must_use]
14pub fn app(
15    kernel: web::Data<HttpKernel>,
16) -> App<
17    impl ServiceFactory<
18        ServiceRequest,
19        Config = (),
20        Response = ServiceResponse<impl MessageBody>,
21        Error = Error,
22        InitError = (),
23    >,
24> {
25    App::new()
26        .app_data(kernel)
27        .default_service(web::to(serenade_service))
28}
29
30/// Binds `addr` and returns the Actix [`actix_web::dev::Server`] (does not await).
31///
32/// Prefer [`listen`] for the usual "run until stop" path. Use this when the caller
33/// needs a [`actix_web::dev::ServerHandle`] for graceful shutdown in tests.
34///
35/// # Errors
36///
37/// Propagates bind errors.
38pub fn bind_server(
39    addr: impl ToSocketAddrs,
40    kernel: HttpKernel,
41) -> std::io::Result<actix_web::dev::Server> {
42    let data = web::Data::new(kernel);
43    Ok(HttpServer::new(move || app(data.clone())).bind(addr)?.run())
44}
45
46/// Awaits a server from [`bind_server`] until it stops.
47///
48/// # Errors
49///
50/// Propagates server IO errors.
51#[allow(clippy::future_not_send)]
52pub async fn await_bound(server: actix_web::dev::Server) -> std::io::Result<()> {
53    server.await
54}
55
56/// Binds `addr` and serves every request through `kernel`.
57///
58/// Prefer this over hand-rolling `HttpServer::bind` in app skeletons.
59///
60/// # Errors
61///
62/// Propagates bind and server IO errors.
63///
64/// Actix's server future is intentionally `!Send` (same as typical Actix handlers).
65#[allow(clippy::future_not_send)]
66pub async fn listen(addr: impl ToSocketAddrs, kernel: HttpKernel) -> std::io::Result<()> {
67    await_bound(bind_server(addr, kernel)?).await
68}
69
70/// Actix `HttpRequest` is `!Send`; clippy nursery flags the resulting future.
71#[allow(clippy::future_not_send)]
72async fn serenade_service(
73    request: HttpRequest,
74    body: web::Bytes,
75    kernel: web::Data<HttpKernel>,
76) -> HttpResponse {
77    dispatch(kernel.get_ref(), &request, body)
78}
