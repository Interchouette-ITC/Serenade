1//! Run registered Serenade HTML apps.
2
3use std::process::{Child, Command};
4
5use anyhow::{Context, bail};
6use clap::Parser;
7use serenade_cli::mode::{discover_apps, find_examples_root, select_apps};
8
9#[derive(Debug, Parser)]
10#[command(name = "serenade-html", about = "Run a registered Serenade HTML app")]
11struct Cli {
12    /// Run every app registered with mode = "html".
13    #[arg(long, conflicts_with = "app")]
14    all: bool,
15    /// Run the app named APP (defaults to demo-html).
16    app: Option<String>,
17}
18
19fn main() -> anyhow::Result<()> {
20    let cli = Cli::parse();
21    let root = find_examples_root(&std::env::current_dir()?)?;
22    let apps = discover_apps(&root.join("examples"))?;
23    let selected = select_apps(&apps, "html", "demo-html", cli.all, cli.app.as_deref())?;
24    let mut children = Vec::with_capacity(selected.len());
25
26    for app in selected {
27        println!("Serving {} ({})", app.name, app.directory.display());
28        let child = match Command::new("cargo")
29            .args(["run", "--manifest-path"])
30            .arg(app.directory.join("Cargo.toml"))
31            .current_dir(&root)
32            .spawn()
33        {
34            Ok(child) => child,
35            Err(error) => {
36                terminate_remaining(&mut children);
37                return Err(error).with_context(|| format!("failed to start app `{}`", app.name));
38            }
39        };
40        children.push((app.name.as_str(), child));
41    }
42
43    wait_for_apps(children)
44}
45
46fn wait_for_apps(mut children: Vec<(&str, Child)>) -> anyhow::Result<()> {
47    while !children.is_empty() {
48        let mut index = 0;
49        while index < children.len() {
50            match children[index]
51                .1
52                .try_wait()
53                .context("failed to check app process")?
54            {
55                Some(status) if status.success() => {
56                    children.remove(index);
57                }
58                Some(status) => {
59                    let (name, _failed_child) = children.remove(index);
60                    terminate_remaining(&mut children);
61                    bail!("HTML app `{name}` exited with status {status}");
62                }
63                None => index += 1,
64            }
65        }
66        if !children.is_empty() {
67            std::thread::sleep(std::time::Duration::from_millis(100));
68        }
69    }
70    Ok(())
71}
72
73fn terminate_remaining(children: &mut [(&str, Child)]) {
74    for (name, child) in children.iter_mut() {
75        if let Err(error) = child.kill() {
76            eprintln!("failed to stop HTML app `{name}`: {error}");
77        }
78    }
79    for (name, child) in children.iter_mut() {
80        if let Err(error) = child.wait() {
81            eprintln!("failed to wait for HTML app `{name}`: {error}");
82        }
83    }
84}
