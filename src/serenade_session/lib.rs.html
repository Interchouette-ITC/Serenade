48    fn bag_set_get_remove_clear() {
49        let mut session = Session::new("abc");
50        assert!(session.is_new());
51        assert_eq!(session.id(), "abc");
52        session.set("user", "42");
53        assert_eq!(session.get("user"), Some("42"));
54        assert!(!session.attributes().is_empty());
55        assert_eq!(session.remove("user"), Some(String::from("42")));
56        assert!(session.get("user").is_none());
57        assert!(session.remove("missing").is_none());
58        session.clear();
59        session.set("a", "1");
60        session.clear();
61        assert!(session.get("a").is_none());
62    }

64    #[test]
65    fn memory_store_roundtrip_and_delete() {
66        let store = MemorySessionStore::new();
67        let mut map = std::collections::HashMap::new();
68        map.insert(String::from("k"), String::from("v"));
69        store.save("s1", &map).expect("save");
70        assert_eq!(
71            store.load("s1").expect("load").as_ref().map(|m| m.get("k")),
72            Some(Some(&String::from("v")))
73        );
74        store.delete("s1").expect("delete");
75        assert!(store.load("s1").expect("load").is_none());
76        assert!(matches!(
77            store.load(""),
78            Err(SessionError::InvalidId { .. })
79        ));
80    }

82    #[test]
83    fn cookie_session_open_commit_reload() {
84        let store = Arc::new(MemorySessionStore::new());
85        let cookies = CookieSession::with_options(
86            store,
87            CookieSessionOptions::new()
88                .with_name("SID")
89                .with_secure(true)
90                .with_same_site(SameSite::Strict)
91                .with_max_age_secs(Some(3600)),
92        );
93        let mut session = cookies.open(None).expect("open");
94        assert!(session.is_new());
95        session.set("cart", "1");
96        let set_cookie = cookies.commit(&session).expect("commit").expect("header");
97        assert!(set_cookie.starts_with("SID="));
98        assert!(set_cookie.contains("HttpOnly"));
99        assert!(set_cookie.contains("Secure"));
100        assert!(set_cookie.contains("SameSite=Strict"));
101        assert!(set_cookie.contains("Max-Age=3600"));
102
103        let id = session.id().to_owned();
104        let header = format!("SID={id}; other=x");
105        let loaded = cookies.open(Some(&header)).expect("reopen");
106        assert!(!loaded.is_new());
107        assert_eq!(loaded.get("cart"), Some("1"));
108        assert!(cookies.commit(&loaded).expect("noop").is_none());
109    }

111    #[test]
112    fn invalidate_clears_store_and_cookie() {
113        let store = Arc::new(MemorySessionStore::new());
114        let cookies = CookieSession::new(Arc::clone(&store) as Arc<dyn SessionStore>);
115        let mut session = cookies.open(None).expect("open");
116        session.set("x", "1");
117        cookies.commit(&session).expect("commit").expect("set");
118        let header = format!("{DEFAULT_SESSION_COOKIE}={}", session.id());
119        let mut again = cookies.open(Some(&header)).expect("load");
120        assert_eq!(again.get("x"), Some("1"));
121        again.invalidate();
122        let clear = cookies.commit(&again).expect("clear").expect("expire");
123        assert!(clear.contains("Max-Age=0"));
124        assert!(store.load(session.id()).expect("load").is_none());
125    }

159    fn same_site_none_and_path() {
160        let store = Arc::new(MemorySessionStore::new());
161        let cookies = CookieSession::with_options(
162            store,
163            CookieSessionOptions::new()
164                .with_path("/app")
165                .with_http_only(false)
166                .with_same_site(SameSite::None)
167                .with_secure(true),
168        );
169        let mut session = cookies.open(None).expect("open");
170        session.set("k", "v");
171        let header = cookies.commit(&session).expect("commit").expect("set");
172        assert!(header.contains("Path=/app"));
173        assert!(!header.contains("HttpOnly"));
174        assert!(header.contains("SameSite=None"));
