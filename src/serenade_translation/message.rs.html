1//! ICU `MessageFormat` subset: `{name}` placeholders and `{count, plural, …}`.
2
3use crate::Locale;
4
5/// Formats a catalogue template with named parameters.
6///
7/// Supports:
8/// - Simple placeholders: `Hello {name}`
9/// - ICU plural blocks: `{count, plural, one {# item} other {# items}}`
10/// - Exact matches: `=0 {none}` inside a plural block
11///
12/// Unknown placeholders are left unchanged. `#` inside a plural arm is replaced
13/// with the absolute plural number.
14#[must_use]
15pub fn format_message(
16    template: &str,
17    parameters: &[(&str, &str)],
18    locale: &Locale,
19    plural_number: Option<i64>,
20) -> String {
21    let mut out = String::with_capacity(template.len());
22    let mut rest = template;
23    while let Some(start) = rest.find('{') {
24        out.push_str(&rest[..start]);
25        let after = &rest[start + 1..];
26        let Some(end) = find_matching_brace(after) else {
27            out.push('{');
28            rest = after;
29            continue;
30        };
31        let inner = &after[..end];
32        rest = &after[end + 1..];
33        out.push_str(&expand_placeholder(
34            inner,
35            parameters,
36            locale,
37            plural_number,
38        ));
39    }
40    out.push_str(rest);
41    out
42}

44fn param<'a>(parameters: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
45    parameters
46        .iter()
47        .find(|(key, _)| *key == name)
48        .map(|(_, value)| *value)
49}
50
51fn find_matching_brace(input: &str) -> Option<usize> {
52    let mut depth = 0_usize;
53    for (idx, ch) in input.char_indices() {
54        match ch {
55            '{' => depth += 1,
56            '}' if depth == 0 => return Some(idx),
57            '}' => depth -= 1,
58            _ => {}
59        }
60    }
61    None
62}

64fn expand_placeholder(
65    inner: &str,
66    parameters: &[(&str, &str)],
67    locale: &Locale,
68    plural_number: Option<i64>,
69) -> String {
70    let trimmed = inner.trim();
71    if let Some((name, rest)) = trimmed.split_once(',') {
72        let name = name.trim();
73        let rest = rest.trim();
74        if let Some(plural_body) = rest.strip_prefix("plural") {
75            let plural_body = plural_body.trim().trim_start_matches(',').trim();
76            let number = plural_number
77                .or_else(|| param(parameters, name).and_then(|v| v.parse().ok()))
78                .unwrap_or(0);
79            return expand_plural(plural_body, number, parameters, locale);
80        }
81    }
82    param(parameters, trimmed).map_or_else(|| format!("{{{trimmed}}}"), ToOwned::to_owned)
83}

85fn expand_plural(body: &str, number: i64, parameters: &[(&str, &str)], locale: &Locale) -> String {
86    let category = plural_category(locale, number);
87    let arms = parse_plural_arms(body);
88    let exact = format!("={number}");
89    let chosen = arms
90        .iter()
91        .find(|(key, _)| key == &exact)
92        .or_else(|| arms.iter().find(|(key, _)| key == &category))
93        .or_else(|| arms.iter().find(|(key, _)| key == "other"))
94        .map_or(body, |(_, text)| text.as_str());
95
96    let with_hash = chosen.replace('#', &number.abs().to_string());
97    format_message(&with_hash, parameters, locale, Some(number))
98}

100/// Parses ICU plural arms (`one {…} other {…}`).
101#[must_use]
102pub fn parse_plural_arms(body: &str) -> Vec<(String, String)> {
103    let mut arms = Vec::new();
104    let mut rest = body.trim();
105    while !rest.is_empty() {
106        let rest_trim = rest.trim_start();
107        let (key, after_key) = match rest_trim.find('{') {
108            Some(idx) => (rest_trim[..idx].trim().to_owned(), &rest_trim[idx + 1..]),
109            None => break,
110        };
111        let Some(end) = find_matching_brace(after_key) else {
112            break;
113        };
114        let value = after_key[..end].to_owned();
115        arms.push((key, value));
116        rest = &after_key[end + 1..];
117    }
118    arms
119}

121#[cfg(feature = "icu")]
122fn plural_category(locale: &Locale, number: i64) -> String {
123    use icu_locale::Locale as IcuLocale;
124    use icu_plurals::{PluralCategory, PluralRuleType, PluralRules, PluralRulesOptions};
125
126    let icu_locale = locale
127        .as_str()
128        .parse::<IcuLocale>()
129        .unwrap_or_else(|_| "en".parse().expect("en"));
130    let options = PluralRulesOptions::from(PluralRuleType::Cardinal);
131    let rules = PluralRules::try_new(icu_locale.into(), options).expect("icu plural data");
132    let category = rules.category_for(number.unsigned_abs());
133    match category {
134        PluralCategory::Zero => "zero",
135        PluralCategory::One => "one",
136        PluralCategory::Two => "two",
137        PluralCategory::Few => "few",
138        PluralCategory::Many => "many",
139        PluralCategory::Other => "other",
140    }
141    .to_owned()
142}

144#[cfg(not(feature = "icu"))]
145fn plural_category(_locale: &Locale, number: i64) -> String {
146    if number == 1 || number == -1 {
147        "one".into()
148    } else {
149        "other".into()
150    }
