3use std::fmt::{Display, Formatter};
4use std::str::FromStr;
5
6use crate::TranslationError;

8/// Language tag (`en`, `fr`, `fr-FR`, …).
9///
10/// Tags are normalized to lowercase language and uppercase region when present
11/// (`fr-fr` → `fr-FR`), matching common Symfony / ICU habits.
12#[derive(Clone, Debug, Eq, Hash, PartialEq, Ord, PartialOrd)]
13pub struct Locale {
14    tag: String,
15}
16
17impl Locale {
18    /// Parses and normalizes a language tag.
19    ///
20    /// # Errors
21    ///
22    /// Returns [`TranslationError::InvalidLocale`] when `raw` is empty or has an
23    /// empty language subtag.
24    pub fn new(raw: impl AsRef<str>) -> Result<Self, TranslationError> {
25        let raw = raw.as_ref().trim();
26        if raw.is_empty() {
27            return Err(TranslationError::InvalidLocale {
28                tag: raw.to_owned(),
29            });
30        }
31        let mut parts = raw.split('-');
32        let language = parts.next().unwrap_or("").trim().to_ascii_lowercase();
33        if language.is_empty() || !language.chars().all(|c| c.is_ascii_alphabetic()) {
34            return Err(TranslationError::InvalidLocale {
35                tag: raw.to_owned(),
36            });
37        }
38        let mut tag = language;
39        if let Some(region) = parts.next() {
40            let region = region.trim();
41            if !region.is_empty() {
42                if !region.chars().all(|c| c.is_ascii_alphanumeric()) {
43                    return Err(TranslationError::InvalidLocale {
44                        tag: raw.to_owned(),
45                    });
46                }
47                tag.push('-');
48                tag.push_str(&region.to_ascii_uppercase());
49            }
50        }
51        Ok(Self { tag })
52    }
53
54    /// Full normalized tag.
55    #[must_use]
56    pub fn as_str(&self) -> &str {
57        &self.tag
58    }
59
60    /// Primary language subtag (`fr` for `fr-FR`).
61    #[must_use]
62    pub fn language(&self) -> &str {
63        self.tag.split('-').next().unwrap_or(self.tag.as_str())
64    }
65
66    /// Region subtag when present (`FR` for `fr-FR`).
67    #[must_use]
68    pub fn region(&self) -> Option<&str> {
69        self.tag.split('-').nth(1)
70    }
71
72    /// Fallback chain excluding `self`: `fr-FR` → `fr`.
73    #[must_use]
74    pub fn parent_chain(&self) -> Vec<Self> {
75        let mut chain = Vec::new();
76        if self.region().is_some() {
77            if let Ok(parent) = Self::new(self.language()) {
78                chain.push(parent);
79            }
80        }
81        chain
82    }
83}
84
85impl Display for Locale {
86    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
87        f.write_str(&self.tag)
88    }
89}
90
91impl FromStr for Locale {
92    type Err = TranslationError;
93
94    fn from_str(s: &str) -> Result<Self, Self::Err> {
95        Self::new(s)
96    }
97}
98
99impl AsRef<str> for Locale {
100    fn as_ref(&self) -> &str {
101        self.as_str()
102    }
