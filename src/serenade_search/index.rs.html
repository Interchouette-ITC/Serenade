3use crate::{SearchDocument, SearchError, SearchHit, SearchQuery};
4
5/// Upsert / delete / query surface for searchable documents.
6///
7/// Adapters own ranking. Applications map domain entities into
8/// [`SearchDocument`] and call this trait after writes.
9///
10/// # Examples
11///
12/// ```
13/// use serenade_search::{DocumentIndex, MemorySearchAdapter, SearchDocument, SearchQuery};
14///
15/// let index = MemorySearchAdapter::new();
16/// index
17///     .upsert(SearchDocument::new("1").field("body", "hello serenade"))
18///     .expect("upsert");
19/// let hits = index.query(&SearchQuery::new("serenade")).expect("query");
20/// assert_eq!(hits[0].id(), "1");
21/// ```
22pub trait DocumentIndex: Send + Sync {
23    /// Inserts or replaces `document` by id.
24    ///
25    /// # Errors
26    ///
27    /// Returns [`SearchError::InvalidId`] when the id is empty, or
28    /// [`SearchError::Index`] when the adapter cannot store the document.
29    fn upsert(&self, document: SearchDocument) -> Result<(), SearchError>;
30
31    /// Removes `id` when present. Returns `true` when a document was deleted.
32    ///
33    /// # Errors
34    ///
35    /// Returns [`SearchError::InvalidId`] when the id is empty, or
36    /// [`SearchError::Index`] when delete fails.
37    fn delete(&self, id: &str) -> Result<bool, SearchError>;
38
39    /// Ranks documents that match `query`.
40    ///
41    /// # Errors
42    ///
43    /// Returns [`SearchError::InvalidQuery`] when the query is rejected, or
44    /// [`SearchError::Index`] when the adapter cannot run the query.
45    fn query(&self, query: &SearchQuery) -> Result<Vec<SearchHit>, SearchError>;
46
47    /// Removes every document.
48    ///
49    /// # Errors
50    ///
51    /// Returns [`SearchError::Index`] when clear fails.
52    fn clear(&self) -> Result<(), SearchError>;
53}
