12/// Environment variable set by `serenade-ssg` for the static output directory.
13pub const STATIC_OUTPUT_DIR_ENV: &str = "SERENADE_STATIC_OUTPUT_DIR";
14
15/// Writes complete documents to route-derived `index.html` files.
16///
17/// Applications enumerate routes and provide page data; this writer owns the
18/// filesystem mapping from each route to its static output path. This
19/// server-side API is omitted from `wasm32` builds.
20#[derive(Clone, Debug, Eq, PartialEq)]
21pub struct StaticDocumentWriter {
22    output_directory: PathBuf,
23}
24
25impl StaticDocumentWriter {
26    /// Creates a writer rooted at `output_directory`.
27    #[must_use]
28    pub fn new(output_directory: impl Into<PathBuf>) -> Self {
29        Self {
30            output_directory: output_directory.into(),
31        }
32    }
33
34    /// Creates a writer using the output directory supplied by `serenade-ssg`.
35    ///
36    /// # Errors
37    ///
38    /// Returns an error when the environment variable is missing or empty.
39    pub fn from_env() -> Result<Self, ViewError> {
40        Self::from_environment(std::env::var_os(STATIC_OUTPUT_DIR_ENV))
41    }
42
43    fn from_environment(output_directory: Option<OsString>) -> Result<Self, ViewError> {
44        let output_directory = output_directory
45            .filter(|directory| !directory.is_empty())
46            .ok_or_else(|| {
47                ViewError::msg(format!(
48                    "environment variable {STATIC_OUTPUT_DIR_ENV} is not set"
49                ))
50            })?;
51        Ok(Self::new(output_directory))
52    }
53
54    /// Resolves a normalized URL route to its static `index.html` path.
55    ///
56    /// `/` maps to `<output>/index.html`; `/about` and `/about/` map to
57    /// `<output>/about/index.html`. Query strings, fragments, and unsafe path
58    /// segments are rejected.
59    ///
60    /// # Errors
61    ///
62    /// Returns an error when `route` is not a safe absolute URL path.
63    pub fn output_path(&self, route: &str) -> Result<PathBuf, ViewError> {
64        Ok(self.index_path(&route_segments(route)?))
65    }
66
67    /// Writes a rendered document at the static path for `route`.
68    ///
69    /// Route directories below the output directory are created one level at a
70    /// time and must be real directories: a symbolic link there is rejected, so
71    /// a document is never written outside the output directory. The document
72    /// is written to a uniquely named temporary file that then replaces
73    /// `index.html` (on Unix and Windows), so an existing symbolic link is
74    /// replaced instead of written through and concurrent writes of the same
75    /// route do not interfere; the last completed write wins.
76    ///
77    /// # Errors
78    ///
79    /// Returns an error when the route is unsafe, a route directory is a
80    /// symbolic link or not a directory, or the document cannot be written.
81    pub fn write(&self, route: &str, document: RenderedDocument) -> Result<PathBuf, ViewError> {
82        let segments = route_segments(route)?;
83        let output_path = self.index_path(&segments);
84        let mut directory = fs::create_dir_all(&self.output_directory)
85            .and_then(|()| self.output_directory.canonicalize())
86            .map_err(|error| directory_error(&self.output_directory, &error))?;
87        for segment in segments {
88            directory.push(segment);
89            create_real_directory(&directory)?;
90        }
91        replace_file(&directory, document.into_static_output().as_bytes()).map_err(|error| {
92            ViewError::msg(format!(
93                "failed to write static document {}: {error}",
94                output_path.display()
95            ))
96        })?;
97        Ok(output_path)
98    }
99}

101impl StaticDocumentWriter {
102    fn index_path(&self, segments: &[&str]) -> PathBuf {
103        let mut output_path = self.output_directory.clone();
104        output_path.extend(segments);
105        output_path.push(INDEX_FILE);
106        output_path
107    }
108}
109
110const INDEX_FILE: &str = "index.html";
111
112fn route_segments(route: &str) -> Result<Vec<&str>, ViewError> {
113    if !route.starts_with('/') || route.contains(['?', '#', '\\']) || route.contains("//") {
114        return Err(ViewError::msg(format!(
115            "static output route must be a normalized absolute path: `{route}`"
116        )));
117    }
118    let relative = route.trim_start_matches('/');
119    let relative = relative.strip_suffix('/').unwrap_or(relative);
120    if relative.is_empty() {
121        return Ok(Vec::new());
122    }
123    relative
124        .split('/')
125        .map(|segment| {
126            let decoded_dots = segment.to_ascii_lowercase().replace("%2e", ".");
127            if segment.is_empty()
128                || matches!(decoded_dots.as_str(), "." | "..")
129                || decoded_dots.contains("%2f")
130                || decoded_dots.contains("%5c")
131                || segment.contains(':')
132                || segment.chars().any(char::is_control)
133            {
134                Err(ViewError::msg(format!(
135                    "unsafe static output route segment in `{route}`"
136                )))
137            } else {
138                Ok(segment)
139            }
140        })
141        .collect()
142}
143
144/// Creates `path` if it is missing; an existing entry must be a real directory.
145/// Concurrent writers may create the same directory, so `AlreadyExists` is
146/// re-checked instead of treated as a failure.
147fn create_real_directory(path: &Path) -> Result<(), ViewError> {
148    match fs::create_dir(path) {
149        Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => {
150            match fs::symlink_metadata(path) {
151                Ok(metadata) if metadata.file_type().is_symlink() => Err(ViewError::msg(format!(
152                    "static output directory {} is a symbolic link; refusing to write through it",
153                    path.display()
154                ))),
155                Ok(metadata) if metadata.is_dir() => Ok(()),
156                _ => Err(directory_error(
157                    path,
158                    &std::io::Error::other("not a directory"),
159                )),
160            }
161        }
162        result => result.map_err(|error| directory_error(path, &error)),
163    }
164}
165
166fn directory_error(path: &Path, error: &std::io::Error) -> ViewError {
167    ViewError::msg(format!(
168        "failed to create static output directory {}: {error}",
169        path.display()
170    ))
171}

173/// Writes `contents` to a uniquely named temporary file in `directory`, then
174/// replaces `index.html` with it. Each write gets its own exclusively created
175/// temporary file, so concurrent writers never share one, and `persist` replaces
176/// an existing destination (including a symbolic link, which is not followed)
177/// on Unix and Windows. A failed write removes its temporary file.
178fn replace_file(directory: &Path, contents: &[u8]) -> std::io::Result<()> {
179    let destination = directory.join(INDEX_FILE);
180    tempfile::Builder::new()
181        .prefix(".index.html.")
182        .suffix(".serenade-tmp")
183        .tempfile_in(directory)
184        .and_then(|mut file| file.write_all(contents).map(|()| file))
185        .and_then(|file| {
186            file.persist(&destination)
187                .map(drop)
188                .or_else(|error| retry_replacement(error, &destination))
189        })
190}
191
192/// On Windows, `persist` uses `MoveFileExW(MOVEFILE_REPLACE_EXISTING)`, which
193/// replaces an existing `index.html` but refuses a read-only one. Clear the
194/// read-only flag on a regular file (never through a symbolic link) and retry
195/// once, so reruns replace previous output as they do on Unix.
196#[cfg(windows)]
197// Windows-only: clears the read-only attribute; the Unix world-writable concern does not apply.
198#[allow(clippy::permissions_set_readonly_false)]
199fn retry_replacement(error: tempfile::PersistError, destination: &Path) -> std::io::Result<()> {
200    let tempfile::PersistError { error, file } = error;
201    match fs::symlink_metadata(destination) {
202        Ok(metadata) if metadata.is_file() && metadata.permissions().readonly() => {
203            let mut permissions = metadata.permissions();
204            permissions.set_readonly(false);
205            fs::set_permissions(destination, permissions)?;
206            file.persist(destination)
207                .map(drop)
208                .map_err(|error| error.error)
209        }
210        _ => Err(error),
211    }
212}
213
214/// Unix `rename` already replaces any existing destination entry.
215#[cfg(not(windows))]
216fn retry_replacement(error: tempfile::PersistError, _destination: &Path) -> std::io::Result<()> {
217    Err(error.error)
218}

335    fn entries(directory: &Path) -> Vec<String> {
336        let mut names: Vec<String> = fs::read_dir(directory)
337            .expect("read directory")
338            .map(|entry| {
339                entry
340                    .expect("entry")
341                    .file_name()
342                    .to_string_lossy()
343                    .into_owned()
344            })
345            .collect();
346        names.sort();
347        names
348    }

409    #[test]
410    #[cfg(unix)]
411    fn never_writes_through_symbolic_links_out_of_the_output_directory() {
412        use std::os::unix::fs::symlink;
413
414        let outside = tempfile::tempdir().expect("outside directory");
415        let temp = tempfile::tempdir().expect("temporary directory");
416        let writer = StaticDocumentWriter::new(temp.path());
417
418        symlink(outside.path(), temp.path().join("about")).expect("directory symlink");
419        let error = writer
420            .write("/about/team", document())
421            .expect_err("symlinked route directory must be rejected")
422            .to_string();
423        assert!(error.contains("is a symbolic link"));
424        assert!(!outside.path().join("team").exists());
425
426        let victim = outside.path().join("victim.html");
427        fs::write(&victim, "untouched").expect("outside file");
428        symlink(&victim, temp.path().join(INDEX_FILE)).expect("file symlink");
429        writer
430            .write("/", document())
431            .expect("replace symlinked document");
432        assert_eq!(fs::read_to_string(&victim).expect("victim"), "untouched");
433        let written = temp.path().join(INDEX_FILE);
434        assert!(
435            !fs::symlink_metadata(&written)
436                .expect("metadata")
437                .file_type()
438                .is_symlink()
439        );
440        assert!(
441            fs::read_to_string(written)
442                .expect("output")
443                .contains("About page")
444        );
445
446        let linked_root = outside.path().join("linked-output");
447        symlink(temp.path(), &linked_root).expect("output directory symlink");
448        StaticDocumentWriter::new(&linked_root)
449            .write("/", document())
450            .expect("a symlinked output directory itself is the configured root");
451    }
