12#[derive(Clone)]
13struct HeadRegistry {
14    entries: RwSignal<Vec<HeadEntry>>,
15    next_id: Arc<AtomicU64>,
16}
17
18impl HeadRegistry {
19    fn new() -> Self {
20        Self {
21            entries: RwSignal::new(Vec::new()),
22            next_id: Arc::new(AtomicU64::new(1)),
23        }
24    }
25
26    fn register(&self, key: Option<String>, content: HeadEntryContent) -> u64 {
27        let id = self.next_id.fetch_add(1, Ordering::Relaxed);
28        self.entries
29            .update(|entries| entries.push(HeadEntry { id, key, content }));
30        id
31    }
32
33    fn remove(&self, id: u64) {
34        self.entries
35            .update(|entries| entries.retain(|entry| entry.id != id));
36    }
37
38    fn visible_entries(&self) -> Vec<HeadEntry> {
39        let entries = self.entries.get();
40        entries
41            .iter()
42            .enumerate()
43            .filter(|(index, entry)| {
44                entry.key.as_ref().is_none_or(|key| {
45                    !entries[index + 1..]
46                        .iter()
47                        .any(|later| later.key.as_ref() == Some(key))
48                })
49            })
50            .map(|(_, entry)| entry.clone())
51            .collect()
52    }
53}

95/// Provides a Leptos metadata context and a lifecycle-managed head insertion point.
96///
97/// Place this component inside the application's `<head>` and supply the active
98/// head view as children. Entries with the same stable key replace earlier
99/// entries while mounted; unkeyed repeatable entries remain.
100#[component]
101pub fn ManagedHead(children: Children) -> impl IntoView {
102    leptos_meta::provide_meta_context();
103    let registry = HeadRegistry::new();
104    provide_context(registry.clone());
105    let children = children();
106
107    view! {
108        // Render nested branches before enumerating entries so SSR includes their registrations.
109        {children}
110        <leptos_meta::MetaTags />
111        <For
112            each=move || registry.visible_entries()
113            key=|entry| entry.id
114            let:entry
115        >
116            <HeadEntryView entry=entry />
117        </For>
118    }
119}

121/// Registers a reactive metadata element and replaces an earlier entry with the same key.
122#[component]
123pub fn ManagedMeta(
124    #[prop(optional, into)] head_key: Option<String>,
125    #[prop(optional, into)] charset: Option<TextProp>,
126    #[prop(optional, into)] name: Option<TextProp>,
127    #[prop(optional, into)] property: Option<TextProp>,
128    #[prop(optional, into)] http_equiv: Option<TextProp>,
129    #[prop(optional, into)] itemprop: Option<TextProp>,
130    #[prop(optional, into)] content: Option<TextProp>,
131) -> impl IntoView {
132    let entry = HeadEntryContent::Meta {
133        charset,
134        name,
135        property,
136        http_equiv,
137        itemprop,
138        content,
139    };
140    if let Some(registry) = use_context::<HeadRegistry>() {
141        let id = registry.register(head_key, entry);
142        on_cleanup(move || registry.remove(id));
143        ().into_any()
144    } else {
145        head_entry_view(HeadEntry::from(entry))
146    }
147}

212fn head_entry_view(entry: HeadEntry) -> AnyView {
213    match entry.content {
214        HeadEntryContent::Title { text } => managed_title_view(text),
215        HeadEntryContent::Meta {
216            charset,
217            name,
218            property,
219            http_equiv,
220            itemprop,
221            content,
222        } => view! {
223            <meta
224                data-serenade-head=server_head_marker()
225                charset=move || charset.as_ref().map(TextProp::get)
226                name=move || name.as_ref().map(TextProp::get)
227                http-equiv=move || http_equiv.as_ref().map(TextProp::get)
228                itemprop=move || itemprop.as_ref().map(TextProp::get)
229                content=move || content.as_ref().map(TextProp::get)
230                {..leptos::tachys::html::attribute::custom::custom_attribute(
231                    "property",
232                    move || property.as_ref().map(TextProp::get)
233                )}
234            />
235        }
236        .into_any(),
237        HeadEntryContent::Link {
238            id,
239            as_,
240            crossorigin,
241            fetchpriority,
242            href,
243            hreflang,
244            imagesizes,
245            imagesrcset,
246            integrity,
247            media,
248            referrerpolicy,
249            rel,
250            sizes,
251            title,
252            type_,
253            blocking,
254        } => view! {
255            <link
256                data-serenade-head=server_head_marker()
257                id=move || id.as_ref().map(TextProp::get)
258                crossorigin=move || crossorigin.as_ref().map(TextProp::get)
259                fetchpriority=move || fetchpriority.as_ref().map(TextProp::get)
260                href=move || href.as_ref().map(TextProp::get)
261                hreflang=move || hreflang.as_ref().map(TextProp::get)
262                imagesizes=move || imagesizes.as_ref().map(TextProp::get)
263                imagesrcset=move || imagesrcset.as_ref().map(TextProp::get)
264                integrity=move || integrity.as_ref().map(TextProp::get)
265                media=move || media.as_ref().map(TextProp::get)
266                referrerpolicy=move || referrerpolicy.as_ref().map(TextProp::get)
267                rel=move || rel.as_ref().map(TextProp::get)
268                sizes=move || sizes.as_ref().map(TextProp::get)
269                title=move || title.as_ref().map(TextProp::get)
270                type=move || type_.as_ref().map(TextProp::get)
271                blocking=move || blocking.as_ref().map(TextProp::get)
272                {..leptos::tachys::html::attribute::custom::custom_attribute(
273                    "as",
274                    move || as_.as_ref().map(TextProp::get)
275                )}
276            />
277        }
278        .into_any(),
279    }
280}

282fn server_head_marker() -> Option<&'static str> {
283    #[cfg(feature = "leptos-ssr")]
284    if use_context::<leptos_meta::ServerMetaContext>().is_some() {
285        return Some("ssr");
286    }
287    None
288}

290/// Uses Leptos' title stack, escaping server-rendered text before metadata injection.
291#[component]
292pub fn ManagedTitle(
293    #[prop(optional, into)] head_key: Option<String>,
294    #[prop(into)] text: TextProp,
295) -> impl IntoView {
296    let entry = HeadEntryContent::Title { text };
297    if let Some(registry) = use_context::<HeadRegistry>() {
298        let id = registry.register(head_key, entry);
299        on_cleanup(move || registry.remove(id));
300        ().into_any()
301    } else {
302        head_entry_view(HeadEntry::from(entry))
303    }
304}
305
306fn managed_title_view(text: TextProp) -> AnyView {
307    #[cfg(feature = "leptos-ssr")]
308    if use_context::<leptos_meta::ServerMetaContext>().is_some() {
309        let title = text.get().to_string();
310        let escaped = crate::escape_html(&title).into_owned();
311        return view! { <leptos_meta::Title text=escaped /> }.into_any();
312    }
313
314    view! { <leptos_meta::Title text=text /> }.into_any()
315}

317/// Emits JSON-LD after JSON serialization and HTML raw-text-safe escaping.
318///
319/// # Panics
320///
321/// Panics while rendering when `json` is not valid JSON. Validate untrusted
322/// input before passing it in.
323#[component]
324pub fn JsonLd(#[prop(into)] json: TextProp) -> impl IntoView {
325    let json = move || {
326        let value: serde_json::Value = serde_json::from_str(&json.get())
327            .unwrap_or_else(|error| panic!("JsonLd requires valid JSON: {error}"));
328        let serialized = serde_json::to_string(&value)
329            .expect("serializing a JSON value to a string cannot fail");
330        serialized
331            .replace('&', "\\u0026")
332            .replace('<', "\\u003c")
333            .replace('>', "\\u003e")
334            .replace('\u{2028}', "\\u2028")
335            .replace('\u{2029}', "\\u2029")
336    };
337
338    view! {
339        <script type="application/ld+json" data-serenade-head=server_head_marker()>{json}</script>
340    }
341}

363    fn removing_route_metadata_restores_the_shared_singleton() {
364        let registry = HeadRegistry::new();
365        let shared = registry.register(
366            Some("meta:name:description".to_owned()),
367            meta_entry("Shared"),
368        );
369        let route = registry.register(
370            Some("meta:name:description".to_owned()),
371            meta_entry("Route"),
372        );
373
374        assert_eq!(
375            registry
376                .visible_entries()
377                .iter()
378                .map(|entry| entry.id)
379                .collect::<Vec<_>>(),
380            [route]
381        );
382
383        registry.remove(route);
384
385        assert_eq!(
386            registry
387                .visible_entries()
388                .iter()
389                .map(|entry| entry.id)
390                .collect::<Vec<_>>(),
391            [shared]
392        );
393    }
