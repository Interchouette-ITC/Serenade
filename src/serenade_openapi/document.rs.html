1//! Info and servers helpers on a utoipa [`OpenApi`](utoipa::openapi::OpenApi) document.
2
3use utoipa::openapi::{Info, OpenApi, Server};
4
5/// Sets `info.title` and `info.version` on `openapi`.
6pub fn apply_info(openapi: &mut OpenApi, title: impl Into<String>, version: impl Into<String>) {
7    openapi.info = Info::new(title.into(), version.into());
8}
9
10/// Sets a single `servers` entry when `server_url` is non-empty after trim.
11///
12/// Pass the public base URL (for example `http://127.0.0.1:8080` or a reverse-proxy
13/// prefix). Empty input clears servers.
14pub fn apply_server(openapi: &mut OpenApi, server_url: impl AsRef<str>) {
15    let url = server_url.as_ref().trim();
16    if url.is_empty() {
17        openapi.servers = None;
18        return;
19    }
20    openapi.servers = Some(vec![Server::new(url)]);
21}
22
23/// Applies title, version, and optional server in one step.
24pub fn finalize_openapi(
25    openapi: &mut OpenApi,
26    title: impl Into<String>,
27    version: impl Into<String>,
28    server_url: Option<&str>,
29) {
30    apply_info(openapi, title, version);
31    match server_url {
32        Some(url) => apply_server(openapi, url),
33        None => openapi.servers = None,
34    }
35}

37#[cfg(test)]
38mod tests {
39    use utoipa::OpenApi;
40
41    use super::*;
42
43    #[derive(OpenApi)]
44    #[openapi(paths())]
45    struct EmptyDoc;
46
47    #[test]
48    fn apply_info_and_server() {
49        let mut doc = EmptyDoc::openapi();
50        finalize_openapi(&mut doc, "demo", "0.1.0", Some("http://127.0.0.1:8080"));
51        assert_eq!(doc.info.title, "demo");
52        assert_eq!(doc.info.version, "0.1.0");
53        let servers = doc.servers.expect("servers");
54        assert_eq!(servers.len(), 1);
55        assert_eq!(servers[0].url, "http://127.0.0.1:8080");
56    }
57
58    #[test]
59    fn finalize_without_server_clears_servers() {
60        let mut doc = EmptyDoc::openapi();
61        apply_server(&mut doc, "http://x");
62        finalize_openapi(&mut doc, "demo", "0.1.0", None);
63        assert_eq!(doc.info.title, "demo");
64        assert!(doc.servers.is_none());
65    }
66
67    #[test]
68    fn apply_info_alone() {
69        let mut doc = EmptyDoc::openapi();
70        apply_info(&mut doc, "only-info", "9.9.9");
71        assert_eq!(doc.info.title, "only-info");
72        assert_eq!(doc.info.version, "9.9.9");
73    }
74
75    #[test]
76    fn apply_server_sets_and_clears() {
77        let mut doc = EmptyDoc::openapi();
78        apply_server(&mut doc, "http://127.0.0.1:8080");
79        assert_eq!(
80            doc.servers.as_ref().expect("servers")[0].url,
81            "http://127.0.0.1:8080"
82        );
83        apply_server(&mut doc, "");
84        assert!(doc.servers.is_none());
85        apply_server(&mut doc, "   ");
86        assert!(doc.servers.is_none());
87    }
88}
