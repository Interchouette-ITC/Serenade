1//! Guest stdout/stderr capture and JSON decode.
2
3use anyhow::{Context, Result, bail};
4use serde::de::DeserializeOwned;
5
6/// Captured guest stdio after a WASIX run.
7#[derive(Debug, Clone, PartialEq, Eq)]
8pub struct GuestOutput {
9    /// Bytes written to stdout.
10    pub stdout: Vec<u8>,
11    /// Bytes written to stderr.
12    pub stderr: Vec<u8>,
13    /// Whether the guest exited successfully.
14    pub success: bool,
15}
16
17/// Decodes JSON from guest stdout when the run succeeded.
18///
19/// # Errors
20///
21/// Returns an error when the guest failed, stdout is empty with stderr present,
22/// or stdout is not valid JSON for `T`.
23pub fn decode_json<T: DeserializeOwned>(output: &GuestOutput) -> Result<T> {
24    if !output.success {
25        bail!(
26            "Wasmer guest failed; stderr: {}; stdout: {}",
27            String::from_utf8_lossy(&output.stderr),
28            String::from_utf8_lossy(&output.stdout)
29        );
30    }
31
32    if output.stdout.is_empty() && !output.stderr.is_empty() {
33        bail!(
34            "guest produced no stdout; stderr: {}",
35            String::from_utf8_lossy(&output.stderr)
36        );
37    }
38
39    serde_json::from_slice(&output.stdout).with_context(|| {
40        format!(
41            "parse guest JSON from stdout `{}` (stderr: {})",
42            String::from_utf8_lossy(&output.stdout),
43            String::from_utf8_lossy(&output.stderr)
44        )
45    })
46}
