26/// Resolves the Wasmer package/module cache root.
27///
28/// Uses `SERENADE_WASMER_CACHE` when set; otherwise `.wasmer` under the current
29/// working directory (products may wrap this with their own default).
30#[must_use]
31pub fn wasmer_cache_root() -> PathBuf {
32    wasmer_cache_root_from(std::env::var_os("SERENADE_WASMER_CACHE"))
33}
34
35/// Resolves cache root from an optional override path.
36#[must_use]
37pub fn wasmer_cache_root_from(override_path: Option<std::ffi::OsString>) -> PathBuf {
38    if let Some(path) = override_path {
39        return PathBuf::from(path);
40    }
41    PathBuf::from(".wasmer")
42}

44/// Downloads (or reads cached) a Wasmer webc package.
45///
46/// # Errors
47///
48/// Returns an error when the cache cannot be read/written or the download fails.
49pub async fn load_webc_from(
50    cache_root: &Path,
51    package_url: &str,
52    cache_name: &str,
53) -> Result<Bytes> {
54    let downloads = cache_root.join("downloads");
55    let cache_path = downloads.join(cache_name);
56    if cache_path.is_file() {
57        return Ok(std::fs::read(&cache_path)
58            .with_context(|| format!("read cached webc {}", cache_path.display()))?
59            .into());
60    }
61    std::fs::create_dir_all(&downloads)
62        .with_context(|| format!("create {}", downloads.display()))?;
63
64    let client = reqwest::Client::builder()
65        .connect_timeout(Duration::from_secs(30))
66        .timeout(Duration::from_secs(300))
67        .build()
68        .context("build HTTP client for Wasmer package download")?;
69    let response = client
70        .get(package_url)
71        .header("Accept", "application/webc")
72        .send()
73        .await
74        .with_context(|| format!("GET {package_url}"))?;
75    if !response.status().is_success() {
76        bail!(
77            "download {package_url} failed with HTTP {}",
78            response.status()
79        );
80    }
81    let body = response.bytes().await.context("read Wasmer webc body")?;
82    ensure_webc_payload(&body)?;
83    std::fs::write(&cache_path, &body)
84        .with_context(|| format!("cache webc at {}", cache_path.display()))?;
85    Ok(body)
86}

88/// Runs a Wasmer registry package command with stdin bytes; returns guest stdio.
89///
90/// # Errors
91///
92/// Returns an error when the package cannot be loaded or the runner task fails
93/// to join (guest failure is reported via [`GuestOutput::success`]).
94pub async fn run_package(request: PackageRun<'_>) -> Result<GuestOutput> {
95    let PackageRun {
96        stdin_bytes,
97        cache_root,
98        package_url,
99        cache_name,
100        command,
101        args,
102    } = request;
103    let webc = load_webc_from(cache_root, package_url, cache_name).await?;
104    let container =
105        from_bytes(webc).with_context(|| format!("decode Wasmer webc {package_url}"))?;
106    let (runtime, tasks) = build_runtime(cache_root)?;
107    let pkg = BinaryPackage::from_webc(&container, &runtime)
108        .await
109        .context("load BinaryPackage from webc")?;
110
111    let stdin = StaticFile::new(OwnedBuffer::from_bytes(stdin_bytes.to_vec()));
112    let mut stdout = virtual_fs::ArcFile::new(Box::<virtual_fs::BufferFile>::default());
113    let mut stderr = virtual_fs::ArcFile::new(Box::<virtual_fs::BufferFile>::default());
114    let stdout_guest = stdout.clone();
115    let stderr_guest = stderr.clone();
116    let runtime = Arc::new(runtime);
117    let command = command.to_owned();
118
119    let join = tokio::task::spawn_blocking(move || {
120        let _guard = tasks.runtime_handle().enter();
121        WasiRunner::new()
122            .with_args(args)
123            .with_stdin(Box::new(stdin) as Box<_>)
124            .with_stdout(Box::new(stdout_guest) as Box<_>)
125            .with_stderr(Box::new(stderr_guest) as Box<_>)
126            .run_command(&command, &pkg, RuntimeOrEngine::Runtime(runtime))
127    });
128
129    let run_result = join.await.context("join wasmer package task")?;
130    read_guest_output(&mut stdout, &mut stderr, run_result.is_ok()).await
131}

133/// Compiles and runs a WASI module from bytes with stdin; returns guest stdio.
134///
135/// # Errors
136///
137/// Returns an error when the module cannot be compiled or the runner task fails
138/// to join.
139pub async fn run_module(
140    program_name: &str,
141    wasm_bytes: &[u8],
142    stdin_bytes: &[u8],
143    cache_root: &Path,
144) -> Result<GuestOutput> {
145    let (runtime, tasks) = build_runtime(cache_root)?;
146    let engine = runtime.engine();
147    let module = Module::new(&engine, wasm_bytes).context("compile WASI module")?;
148    let module_hash = ModuleHash::new(wasm_bytes);
149
150    let stdin = StaticFile::new(OwnedBuffer::from_bytes(stdin_bytes.to_vec()));
151    let mut stdout = virtual_fs::ArcFile::new(Box::<virtual_fs::BufferFile>::default());
152    let mut stderr = virtual_fs::ArcFile::new(Box::<virtual_fs::BufferFile>::default());
153    let stdout_guest = stdout.clone();
154    let stderr_guest = stderr.clone();
155    let program = program_name.to_owned();
156
157    let join = tokio::task::spawn_blocking(move || {
158        let _guard = tasks.runtime_handle().enter();
159        WasiRunner::new()
160            .with_stdin(Box::new(stdin) as Box<_>)
161            .with_stdout(Box::new(stdout_guest) as Box<_>)
162            .with_stderr(Box::new(stderr_guest) as Box<_>)
163            .run_wasm(
164                RuntimeOrEngine::Engine(engine),
165                &program,
166                module,
167                module_hash,
168            )
169    });
170
171    let run_result = join.await.context("join wasmer wasi task")?;
172    read_guest_output(&mut stdout, &mut stderr, run_result.is_ok()).await
173}

175fn ensure_webc_payload(body: &[u8]) -> Result<()> {
176    if body.len() < 1024 {
177        bail!(
178            "Wasmer package download looks too small ({} bytes); check Accept header",
179            body.len()
180        );
181    }
182    Ok(())
183}

185fn build_runtime(cache_root: &Path) -> Result<(PluggableRuntime, Arc<TokioTaskManager>)> {
186    let tasks = Arc::new(TokioTaskManager::new(Handle::current()));
187    let mut runtime = PluggableRuntime::new(Arc::clone(&tasks) as Arc<_>);
188    let compiled = cache_root.join("compiled");
189    let packages = cache_root.join("packages");
190    std::fs::create_dir_all(&compiled).with_context(|| format!("create {}", compiled.display()))?;
191    std::fs::create_dir_all(&packages).with_context(|| format!("create {}", packages.display()))?;
192    let module_cache =
193        SharedCache::default().with_fallback(FileSystemCache::new(compiled, Arc::clone(&tasks)));
194    runtime
195        .set_module_cache(module_cache)
196        .set_package_loader(BuiltinPackageLoader::new().with_cache_dir(packages));
197    Ok((runtime, tasks))
198}

200async fn read_guest_output(
201    stdout: &mut (impl AsyncReadExt + AsyncSeekExt + Unpin),
202    stderr: &mut (impl AsyncReadExt + AsyncSeekExt + Unpin),
203    success: bool,
204) -> Result<GuestOutput> {
205    stdout.rewind().await.context("rewind stdout")?;
206    stderr.rewind().await.context("rewind stderr")?;
207    let mut out_buf = Vec::new();
208    let mut err_buf = Vec::new();
209    stdout
210        .read_to_end(&mut out_buf)
211        .await
212        .context("read stdout")?;
213    stderr
214        .read_to_end(&mut err_buf)
215        .await
216        .context("read stderr")?;
217    Ok(GuestOutput {
218        stdout: out_buf,
219        stderr: err_buf,
220        success,
221    })
