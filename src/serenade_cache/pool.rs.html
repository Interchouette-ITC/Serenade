3use std::any::Any;
4use std::collections::HashMap;
5use std::sync::{Arc, Mutex, MutexGuard};
6use std::time::Instant;
7
8use crate::{ArrayCacheItem, CacheError};
9
10/// Pool that loads and persists [`CacheItem`](crate::CacheItem)s (PSR-6 analogue).
11pub trait CacheItemPool: Send + Sync {
12    /// Returns an item for `key` (miss when absent or expired).
13    ///
14    /// # Errors
15    ///
16    /// Returns [`CacheError`] when the key is invalid or the pool fails.
17    fn get_item(&self, key: &str) -> Result<ArrayCacheItem, CacheError>;
18
19    /// Persists `item`.
20    ///
21    /// # Errors
22    ///
23    /// Returns [`CacheError`] when save fails.
24    fn save(&self, item: ArrayCacheItem) -> Result<(), CacheError>;
25
26    /// Deletes `key` when present.
27    ///
28    /// # Errors
29    ///
30    /// Returns [`CacheError`] when delete fails.
31    fn delete_item(&self, key: &str) -> Result<bool, CacheError>;
32
33    /// Clears all items.
34    ///
35    /// # Errors
36    ///
37    /// Returns [`CacheError`] when clear fails.
38    fn clear(&self) -> Result<(), CacheError>;
39}
40
41struct Stored {
42    value: Option<Arc<dyn Any + Send + Sync>>,
43    expires_at: Option<Instant>,
44}
45
46/// In-memory [`CacheItemPool`] (`ArrayAdapter`).
47#[derive(Default)]
48pub struct ArrayAdapter {
49    inner: Mutex<HashMap<String, Stored>>,
50}
51
52impl ArrayAdapter {
53    /// Creates an empty in-memory pool.
54    #[must_use]
55    pub fn new() -> Self {
56        Self::default()
57    }
58
59    fn validate_key(key: &str) -> Result<(), CacheError> {
60        if key.is_empty() {
61            return Err(CacheError::InvalidKey {
62                key: key.to_owned(),
63                message: "key must not be empty".to_owned(),
64            });
65        }
66        Ok(())
67    }
68
69    fn lock_map(&self) -> MutexGuard<'_, HashMap<String, Stored>> {
70        self.inner
71            .lock()
72            .expect("serenade-cache ArrayAdapter mutex poisoned")
73    }
74}
75
76impl CacheItemPool for ArrayAdapter {
77    fn get_item(&self, key: &str) -> Result<ArrayCacheItem, CacheError> {
78        Self::validate_key(key)?;
79        let mut map = self.lock_map();
80        if map
81            .get(key)
82            .is_some_and(|stored| stored.expires_at.is_some_and(|at| Instant::now() >= at))
83        {
84            map.remove(key);
85            return Ok(ArrayCacheItem::miss(key));
86        }
87        let item = map.get(key).map_or_else(
88            || ArrayCacheItem::miss(key),
89            |stored| {
90                stored.value.as_ref().map_or_else(
91                    || ArrayCacheItem::miss(key),
92                    |value| {
93                        ArrayCacheItem::hit(key, Arc::clone(value)).with_expiry(stored.expires_at)
94                    },
95                )
96            },
97        );
98        drop(map);
99        Ok(item)
100    }
101
102    fn save(&self, item: ArrayCacheItem) -> Result<(), CacheError> {
103        let (key, value, expires_at) = item.into_stored();
104        Self::validate_key(&key)?;
105        self.lock_map().insert(key, Stored { value, expires_at });
106        Ok(())
107    }
108
109    fn delete_item(&self, key: &str) -> Result<bool, CacheError> {
110        Self::validate_key(key)?;
111        Ok(self.lock_map().remove(key).is_some())
112    }
113
114    fn clear(&self) -> Result<(), CacheError> {
115        self.lock_map().clear();
116        Ok(())
117    }
118}
