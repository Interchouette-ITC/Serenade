1//! Symfony-shaped rate limiting.
2//!
3//! - [`Policy`]: token-bucket or fixed-window limits
4//! - [`RateLimiter`] / [`RateLimiterFactory`]: `consume` / `reset`
5//! - [`RateLimiterStorage`] / [`InMemoryRateLimiterStorage`]: persist window state
6//! - [`RateLimit`]: accept / remaining / retry-after
7//! - [`consume_or_exceed`] / [`require_accepted`]: app-facing helpers
8//! - [`too_many_requests`]: HTTP 429 bridge
9//! - [`RegisterDefaultRateLimiterPass`]: DI default service id [`DEFAULT_RATE_LIMITER_STORAGE_SERVICE`]
10//!
11//! # Examples
12//!
13//! ```
14//! use std::sync::Arc;
15//! use std::time::Duration;
16//! use serenade_rate_limiter::{
17//!     InMemoryRateLimiterStorage, Policy, RateLimiterFactory, consume_or_exceed,
18//! };
19//!
20//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
21//! let storage = Arc::new(InMemoryRateLimiterStorage::new());
22//! let factory = RateLimiterFactory::new(
23//!     "login",
24//!     Policy::fixed_window(5, Duration::from_secs(15 * 60))?,
25//!     storage,
26//! )?;
27//! let limiter = factory.create("alice")?;
28//! consume_or_exceed(&limiter, 1)?;
29//! # Ok(())
30//! # }
31//! ```

64/// Rejects empty limiter keys.
65///
66/// # Errors
67///
68/// Returns [`RateLimiterError::InvalidKey`] when `key` is empty.
69pub fn validate_key(key: &str) -> Result<(), RateLimiterError> {
70    if key.is_empty() {
71        return Err(RateLimiterError::InvalidKey {
72            key: key.to_owned(),
73            message: "key must not be empty".to_owned(),
74        });
75    }
76    Ok(())
77}
