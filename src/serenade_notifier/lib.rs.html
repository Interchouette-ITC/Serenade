1//! SMS and push notification channels (Symfony Notifier shaped).
2//!
3//! Types: [`Notification`], [`Channel`], [`SmsMessage`], [`PushMessage`].
4//! Transports: [`NullTransport`], [`MemoryTransport`], [`SmsOnlyTransport`].
5//! DI: [`RegisterDefaultNotifierPass`] seeds [`DEFAULT_NOTIFIER_SERVICE`].
6//!
7//! # Examples
8//!
9//! ```
10//! use serenade_notifier::{MemoryTransport, Notification, Transport};
11//!
12//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
13//! let transport = MemoryTransport::new();
14//! transport.send(&Notification::sms("+15551212", "hello"))?;
15//! assert_eq!(transport.sent().len(), 1);
16//! # Ok(())
17//! # }
18//! ```
