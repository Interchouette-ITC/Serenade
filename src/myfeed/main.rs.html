138fn extract_csrf_hidden(form_html: &str) -> String {
139    let marker = r#"<input type="hidden" name="_token""#;
140    let Some(start) = form_html.find(marker) else {
141        return String::new();
142    };
143    let rest = &form_html[start..];
144    let Some(end) = rest.find("/>") else {
145        return String::new();
146    };
147    rest[..=end + 1].to_owned()
148}

171fn build_composer_form(
172    csrf: &HmacCsrfTokenManager,
173    categories: &[String],
174    form_name: &str,
175    action: &str,
176    post: Option<&Post>,
177    submit_label: &str,
178) -> Result<String, HttpError> {
179    let mut form = Form::builder(form_name)
180        .action(action)
181        .field("body", not_blank())
182        .field("embed_url", vec![])
183        .field("image_data", vec![])
184        .field("category", vec![])
185        .build();
186    form.prepare_csrf(csrf)
187        .map_err(|err| HttpError::failed(err.to_string()))?;
188    let full = form
189        .render()
190        .map_err(|err| HttpError::failed(err.to_string()))?;
191    let csrf_field = extract_csrf_hidden(full.as_html());
192    let emojis = emoji_picker();
193    let selected_category = post.map_or("", |p| p.category.as_str());
194    let mut options = String::new();
195    for (index, name) in categories.iter().enumerate() {
196        let selected = if (!selected_category.is_empty() && name == selected_category)
197            || (selected_category.is_empty() && index == 0)
198        {
199            " selected"
200        } else {
201            ""
202        };
203        let _ = write!(
204            options,
205            r#"<option value="{value}"{selected}>{label}</option>"#,
206            value = escape_html(name),
207            label = escape_html(name),
208        );
209    }
210    if options.is_empty() {
211        options.push_str(r#"<option value="Life">Life</option>"#);
212    }
213    let embed_value = post.and_then(|p| p.embed_url.as_deref()).unwrap_or("");
214    let image_value = post.and_then(|p| p.image_data.as_deref()).unwrap_or("");
215    let initial_attr = post.map_or(String::new(), |p| {
216        format!(
217            r#" data-initial-html="{html}""#,
218            html = escape_attr(&p.body)
219        )
220    });
221    let cancel = if post.is_some() {
222        r#"<a href="/" class="btn btn-outline-secondary">Cancel</a>"#
223    } else {
224        r#"<button type="button" id="composer-cancel" class="btn btn-outline-secondary">Cancel</button>"#
225    };
226    Ok(format!(
227        r#"<form name="{form_name}" method="POST" action="{action}" class="composer-form">
228{csrf_field}
229<input type="hidden" id="body" name="body" value="" />
230<input type="hidden" id="image_data" name="image_data" value="{image_value}" />
231
232<div class="mb-3 composer-emoji-wrap">
233  <div id="composer-quill"{initial_attr}></div>
234  <div class="d-flex justify-content-end mt-1">
235    <span id="composer-count" class="text-secondary small">0/2000</span>
236  </div>
237  {emojis}
238</div>
239
240<div class="mb-3">
241  <label class="form-label" for="media_upload">Media (image) - optional</label>
242  <input type="file" id="media_upload" class="form-control" accept="image/*" />
243  <div id="media-preview" class="mt-2"></div>
244  <label class="form-label mt-3" for="embed_url">…or paste a link instead</label>
245  <input type="url" class="form-control" id="embed_url" name="embed_url" value="{embed_value}" placeholder="https://… (.mp4, YouTube, .jpg)" />
246</div>
247
248<div class="mb-3">
249  <label class="form-label" for="category">Category</label>
250  <select class="form-select" id="category" name="category">
251    {options}
252  </select>
253</div>
254
255<div class="d-flex gap-2 justify-content-end">
256  {cancel}
257  <button type="submit" class="btn btn-primary px-4">{submit_label}</button>
258</div>
259</form>"#,
260        image_value = escape_html(image_value),
261        embed_value = escape_html(embed_value),
262        submit_label = escape_html(submit_label),
263    ))
264}

287fn extract_csrf_value(csrf_hidden: &str) -> String {
288    let key = "value=\"";
289    let Some(start) = csrf_hidden.find(key) else {
290        return String::new();
291    };
292    csrf_hidden[start + key.len()..]
293        .split('"')
294        .next()
295        .unwrap_or("")
296        .to_owned()
297}

453fn cookie_value(request: &Request, name: &str) -> Option<String> {
454    let header = request.headers().get("cookie")?;
455    for part in header.split(';') {
456        let part = part.trim();
457        if let Some((key, value)) = part.split_once('=') {
458            if key.trim() == name {
459                return Some(value.trim().to_owned());
460            }
461        }
462    }
463    None
464}

466fn is_admin(request: &Request, expected: &str) -> bool {
467    if request.headers().get("Authorization").is_some_and(|value| {
468        let key = value.strip_prefix("Bearer ").unwrap_or(value);
469        key == expected
470    }) {
471        return true;
472    }
473    cookie_value(request, ADMIN_COOKIE).is_some_and(|value| value == expected)
474}

476fn ui_for<'a>(state: &'a AppState, request: &Request) -> Ui<'a> {
477    let locale = serenade_translation::request_locale(request)
478        .cloned()
479        .unwrap_or_else(|| state.negotiator.resolve(request));
480    state.translator.set_locale(locale.clone());
481    Ui::new(&state.translator, locale)
482}

484fn handle_locale_set(state: &AppState, request: &Request) -> Response {
485    let code = request
486        .attributes()
487        .get::<String>("code")
488        .map_or("en", String::as_str);
489    let locale = Locale::new(code).unwrap_or_else(|_| Locale::new("en").expect("en"));
490    let matched = state
491        .negotiator
492        .allowed()
493        .iter()
494        .find(|item| item.language() == locale.language())
495        .cloned()
496        .unwrap_or_else(|| state.negotiator.default_locale().clone());
497    let cookie = format!(
498        "{LOCALE_COOKIE}={tag}; Path=/; SameSite=Lax; Max-Age=31536000",
499        tag = matched.as_str()
500    );
501    redirect_with_cookie("/", &cookie)
502}

504fn feed_with_flash(
505    state: &AppState,
506    request: &Request,
507    flash: Option<&str>,
508    is_err: bool,
509    composer_open: bool,
510) -> Result<Response, HttpError> {
511    let ui = ui_for(state, request);
512    let started = Instant::now();
513    let categories = state.store.categories();
514    let posts = state.store.posts();
515    if let Some(token) = request.attributes().get::<String>(PROFILER_TOKEN_ATTRIBUTE) {
516        record_query(
517            &state.profiler,
518            token,
519            QueryEvent::new(
520                "SELECT posts + categories (feed page)",
521                started.elapsed().max(Duration::from_micros(1)),
522            ),
523        );
524    }
525    let post_form = build_post_form(&state.csrf, &categories)?;
526    let mut comment_forms = Vec::new();
527    let mut like_forms = Vec::new();
528    let mut admin_forms = Vec::new();
529    let admin = is_admin(request, &state.admin_token);
530    for post in &posts {
531        comment_forms.push((post.id, build_comment_form(&state.csrf, post.id)?));
532        like_forms.push((post.id, build_like_form(&state.csrf, post.id)?));
533        if admin {
534            admin_forms.push((post.id, build_admin_post_actions(&state.csrf, post.id)?));
535        }
536    }
537    Ok(html_response(
538        if is_err { 400 } else { 200 },
539        feed_page(&FeedView {
540            ui: &ui,
541            store: &state.store,
542            post_form_html: &post_form,
543            comment_forms: &comment_forms,
544            like_forms: &like_forms,
545            admin_forms: &admin_forms,
546            flash,
547            flash_err: is_err,
548            composer_open,
549        }),
550    ))
551}

553fn handle(state: &AppState, request: &mut Request) -> Result<Response, HttpError> {
554    if let Some(response) = try_handle_profiler(&state.profiler, "/_profiler", request) {
555        return Ok(response);
556    }
557    state.matcher.apply(request)?;
558    let _locale = state.negotiator.apply(request);
559    tracing::info!(target: REQUEST, path = request.path(), "myfeed request");
560    let route = request
561        .attributes()
562        .get::<String>(ROUTE_ATTRIBUTE)
563        .map_or("", String::as_str);
564
565    match route {
566        "feed" => feed_with_flash(state, request, None, false, false),
567        "post_create" => handle_post_create(state, request),
568        "comment_create" => handle_comment_create(state, request),
569        "post_like" => handle_like(state, request),
570        "admin" => handle_admin_get(state, request),
571        "admin_login" => handle_admin_login(state, request),
572        "admin_logout" => Ok(handle_admin_logout(state, request)),
573        "admin_approve" => handle_admin_moderation(state, request, true),
574        "admin_reject" => handle_admin_moderation(state, request, false),
575        "admin_category_add" => handle_category_add(state, request),
576        "admin_category_delete" => handle_category_delete(state, request),
577        "admin_post_edit_get" => handle_post_edit_get(state, request),
578        "admin_post_edit_post" => handle_post_edit_post(state, request),
579        "admin_post_delete" => handle_post_delete(state, request),
580        "asset_css" => Ok(asset_response(
581            "text/css; charset=utf-8",
582            include_bytes!("../assets/myfeed.css"),
583        )),
584        "asset_js" => Ok(asset_response(
585            "text/javascript; charset=utf-8",
586            include_bytes!("../assets/clitorine.js"),
587        )),
588        "locale_set" => Ok(handle_locale_set(state, request)),
589        _ => Err(HttpError::not_found("no handler")),
590    }
591}

593fn sanitize_image_data(raw: &str) -> Option<String> {
594    let raw = raw.trim();
595    if raw.is_empty() {
596        return None;
597    }
598    if raw.len() > MAX_IMAGE_DATA {
599        return None;
600    }
601    if !(raw.starts_with("data:image/png;base64,")
602        || raw.starts_with("data:image/jpeg;base64,")
603        || raw.starts_with("data:image/jpg;base64,")
604        || raw.starts_with("data:image/webp;base64,")
605        || raw.starts_with("data:image/gif;base64,"))
606    {
607        return None;
608    }
609    if !raw
610        .chars()
611        .all(|c| c.is_ascii_alphanumeric() || matches!(c, ':' | '/' | ';' | ',' | '+' | '=' | '_'))
612    {
613        return None;
614    }
615    Some(raw.to_owned())
616}

618fn parse_new_post(state: &AppState, form: &Form) -> Result<NewPost, &'static str> {
619    let raw_body = form.get("body").unwrap_or("").trim();
620    let body = sanitize_post_html(raw_body);
621    if plain_len(&body) == 0 || plain_len(&body) > MAX_BODY_CHARS {
622        return Err("Post body is required (max 2000 characters).");
623    }
624    let embed_raw = form.get("embed_url").unwrap_or("").trim();
625    let embed_url = if embed_raw.is_empty() {
626        None
627    } else if is_allowed_embed(embed_raw) {
628        Some(embed_raw.to_owned())
629    } else {
630        return Err("Media URL must be YouTube, SoundCloud, image, or .mp4.");
631    };
632    let image_raw = form.get("image_data").unwrap_or("");
633    let image_data = if image_raw.trim().is_empty() {
634        None
635    } else if let Some(data) = sanitize_image_data(image_raw) {
636        Some(data)
637    } else {
638        return Err("Image upload rejected (type or size).");
639    };
640    let category = form.get("category").unwrap_or("").trim();
641    let category = if state.store.has_category(category) {
642        category.to_owned()
643    } else {
644        state
645            .store
646            .categories()
647            .into_iter()
648            .next()
649            .unwrap_or_else(|| "Life".to_owned())
650    };
651    Ok(NewPost {
652        body,
653        embed_url,
654        image_data,
655        category,
656    })
657}

720                state,
721                request,
722                Some("Comment sent. It will appear after approval."),
723                false,
724                false,
725            )
726        }
727        _ if wants_ajax(request) => Ok(ajax_text("Could not send comment.")),
728        _ => Ok(redirect("/")),
729    }

1022fn main() -> Result<(), Box<dyn std::error::Error>> {
1023    let bind = std::env::var("MYFEED_BIND").unwrap_or_else(|_| DEFAULT_BIND.to_owned());
1024    let csrf_secret =
1025        std::env::var("MYFEED_CSRF_SECRET").unwrap_or_else(|_| DEFAULT_CSRF.to_owned());
1026    let admin_token =
1027        std::env::var("MYFEED_ADMIN_TOKEN").unwrap_or_else(|_| DEFAULT_ADMIN.to_owned());
1028
1029    let db_path = std::env::var("MYFEED_DB").unwrap_or_else(|_| ".myfeed.sqlite".to_owned());
1030    let store = FeedStore::open(db_path);
1031    seed_if_empty(&store);
1032    let translations = PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("translations");
1033    let mut translator = Translator::new(Locale::new("en").expect("en"))
1034        .with_fallbacks(vec![Locale::new("en").expect("en")]);
1035    translator
1036        .load_path(&translations)
1037        .map_err(|err| format!("load translations: {err}"))?;
1038    let negotiator = LocaleNegotiator::new(Locale::new("en").expect("en"))
1039        .with_allowed(vec![
1040            Locale::new("en").expect("en"),
1041            Locale::new("fr").expect("fr"),
1042        ])
1043        .with_cookie_name(LOCALE_COOKIE);
1044
1045    let profiler_enabled = std::env::var("MYFEED_PROFILER").map_or(true, |value| {
1046        value != "0" && !value.eq_ignore_ascii_case("false")
1047    });
1048    let profiler_store = Arc::new(ProfileStore::new(50));
1049    let _ = tracing_subscriber::registry()
1050        .with(ProfilerLogLayer)
1051        .with(tracing_subscriber::fmt::layer().with_target(true))
1052        .try_init();
1053
1054    let state = Arc::new(AppState {
1055        store,
1056        csrf: HmacCsrfTokenManager::new(csrf_secret.as_bytes()),
1057        matcher: UrlMatcher::new(routes()?),
1058        admin_token: admin_token.clone(),
1059        translator,
1060        negotiator,
1061        profiler: Arc::clone(&profiler_store),
1062    });
1063
1064    let state_for_handler = Arc::clone(&state);
1065    let mut async_kernel = AsyncHttpKernel::from_sync(move |request: &mut Request| {
1066        handle(state_for_handler.as_ref(), request)
1067    });
1068    if profiler_enabled {
1069        async_kernel.push_middleware(AsyncProfilerMiddleware::new(
1070            Arc::clone(&profiler_store),
1071            ProfilerConfig::enabled(50),
1072        ));
1073    }
1074
1075    println!("MyFeed listening on http://{bind}/");
1076    println!("Admin: open /admin and sign in with token `{admin_token}`");
1077    if profiler_enabled {
1078        println!("Profiler: http://{bind}/_profiler (disable with MYFEED_PROFILER=0)");
1079    }
1080
1081    let runtime = tokio::runtime::Runtime::new()?;
1082    runtime.block_on(serenade_http_actix::listen(bind, async_kernel))?;
1083    Ok(())
