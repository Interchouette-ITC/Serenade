18/// Full HTML document with Bootstrap, Quill, and Clitorine.
19#[must_use]
20pub fn document(title: &str, body: &str, composer_open: bool) -> String {
21    let open_flag = if composer_open { "1" } else { "0" };
22    format!(
23        r#"<!DOCTYPE html>
24<html lang="en">
25<head>
26<meta charset="utf-8" />
27<meta name="viewport" content="width=device-width, initial-scale=1" />
28<title>{title}</title>
29<link rel="stylesheet" href="{BOOTSTRAP_CSS}" />
30<link rel="stylesheet" href="{QUILL_CSS}" />
31<link rel="stylesheet" href="/assets/myfeed.css?v=5" />
32</head>
33<body class="myfeed" data-composer-open="{open_flag}">
34{body}
35<script src="{BOOTSTRAP_JS}"></script>
36<script src="{QUILL_JS}"></script>
37<script src="/assets/clitorine.js?v=4"></script>
38</body>
39</html>"#,
40        title = escape_html(title),
41    )
42}

133fn render_wall(view: &FeedView<'_>) -> String {
134    let posts = view.store.posts();
135    if posts.is_empty() {
136        return r#"<p class="text-secondary mb-0">No posts yet. Be the first on the wall.</p>"#
137            .to_owned();
138    }
139    let mut posts_html = String::new();
140    for post in &posts {
141        let comments = view.store.approved_comments(post.id);
142        let comment_form = view
143            .comment_forms
144            .iter()
145            .find(|(id, _)| *id == post.id)
146            .map_or("", |(_, html)| html.as_str());
147        let like_form = view
148            .like_forms
149            .iter()
150            .find(|(id, _)| *id == post.id)
151            .map_or("", |(_, html)| html.as_str());
152        let admin_actions = view
153            .admin_forms
154            .iter()
155            .find(|(id, _)| *id == post.id)
156            .map_or("", |(_, html)| html.as_str());
157        posts_html.push_str(&post_card(
158            post,
159            &comments,
160            comment_form,
161            like_form,
162            admin_actions,
163        ));
164    }
165    posts_html
166}

207fn post_card(
208    post: &Post,
209    comments: &[Comment],
210    comment_form: &str,
211    like_form: &str,
212    admin_actions: &str,
213) -> String {
214    let embed = post
215        .embed_url
216        .as_deref()
217        .and_then(embed_html)
218        .unwrap_or_default();
219    let image = post.image_data.as_deref().map_or(String::new(), |data| {
220        if data.starts_with("data:image/") && data.len() < 280_000 {
221            format!(
222                r#"<div class="embed-media my-2"><img class="img-fluid rounded" src="{src}" alt="posted image" loading="lazy" /></div>"#,
223                src = escape_attr(data)
224            )
225        } else {
226            String::new()
227        }
228    });
229    let mut comments_html = String::new();
230    for comment in comments {
231        let _ = write!(
232            comments_html,
233            r#"<p class="mb-2 small">{body}</p>"#,
234            body = escape_html(&comment.body)
235        );
236    }
237    if comments_html.is_empty() {
238        comments_html
239            .push_str(r#"<p class="text-secondary small mb-2">No approved comments yet.</p>"#);
240    }
241    let when = if post.created_at.is_empty() {
242        String::new()
243    } else {
244        format!(" · {}", escape_html(&post.created_at))
245    };
246    format!(
247        r#"
248<article class="card post-card" id="post-{id}">
249  <div class="card-body p-4">
250    <div class="d-flex justify-content-between align-items-start gap-2 mb-2">
251      <p class="text-secondary small mb-0">Post #{id} · {category}{when}</p>
252      <div class="d-flex flex-wrap align-items-center gap-2">
253        <span class="badge text-bg-light border" data-like-count>{likes} likes</span>
254        {like_form}
255        {admin_actions}
256      </div>
257    </div>
258    <div class="post-body mb-2">{body}</div>
259    {image}
260    {embed}
261    <div class="border-top pt-3 mt-3">
262      <h3 class="h6">Comments</h3>
263      {comments_html}
264      {comment_form}
265    </div>
266  </div>
267</article>
268"#,
269        id = post.id,
270        category = escape_html(&post.category),
271        likes = post.likes,
272        body = post.body,
273    )
274}

276/// Admin page with logout, comment queue, and category manager.
277#[must_use]
278pub fn admin_page_with_logout(
279    pending: &[Comment],
280    forms: &[(u64, String, String)],
281    logout_form: &str,
282    categories_html: &str,
283    notice: Option<&str>,
284) -> String {
285    let notice_html = notice.map_or(String::new(), |notice| {
286        format!(
287            r#"<div class="alert alert-success" role="alert">{notice}</div>"#,
288            notice = escape_html(notice)
289        )
290    });
291    let mut rows = String::new();
292    if pending.is_empty() {
293        rows.push_str(r#"<p class="text-secondary mb-0">Queue empty.</p>"#);
294    }
295    for comment in pending {
296        let (approve, reject) = forms
297            .iter()
298            .find(|(id, _, _)| *id == comment.id)
299            .map_or(("", ""), |(_, approve, reject)| {
300                (approve.as_str(), reject.as_str())
301            });
302        let _ = write!(
303            rows,
304            r#"
305<div class="card post-card mb-3">
306  <div class="card-body">
307    <p class="text-secondary small">Comment #{id} on post #{post}</p>
308    <p>{body}</p>
309    <div class="d-flex gap-2">{approve}{reject}</div>
310  </div>
311</div>
312"#,
313            id = comment.id,
314            post = comment.post_id,
315            body = escape_html(&comment.body),
316        );
317    }
318    let body = format!(
319        r#"
320<div class="myfeed-shell">
321  <header class="d-flex flex-wrap align-items-start gap-3 mb-4 pb-3 border-bottom">
322    <div class="me-auto">
323      <h1 class="myfeed-brand">MyFeed Admin</h1>
324      <p class="myfeed-tag">Moderation queue and categories</p>
325    </div>
326    <nav class="myfeed-nav d-flex align-items-center gap-3 pt-2">
327      <a class="link-secondary" href="/">Feed</a>
328      <a class="admin-link btn btn-primary btn-sm" href="/admin">Admin</a>
329      {logout_form}
330    </nav>
331  </header>
332  <main class="row g-4">
333    <div class="col-lg-7">
334      <h2 class="h5 mb-3">Pending comments</h2>
335      {notice_html}
336      {rows}
337    </div>
338    <div class="col-lg-5">
339      <h2 class="h5 mb-3">Categories</h2>
340      {categories_html}
341    </div>
342  </main>
343</div>
344"#
345    );
346    document("MyFeed Admin", &body, false)
347}

417/// HTML response helper.
418#[must_use]
419pub fn html_response(status: u16, body: String) -> serenade_http::Response {
420    serenade_http::Response::new(status)
421        .with_header("content-type", "text/html; charset=utf-8")
422        .with_body(body.into_bytes())
423}
424
425/// Static asset response.
426#[must_use]
427pub fn asset_response(content_type: &str, bytes: &'static [u8]) -> serenade_http::Response {
428    serenade_http::Response::new(200)
429        .with_header("content-type", content_type)
430        .with_header("cache-control", "no-store")
431        .with_body(bytes.to_vec())
432}

434/// Redirect helper (paths are app-controlled).
435#[must_use]
436pub fn redirect(location: &str) -> serenade_http::Response {
437    serenade_http::Response::new(303)
438        .with_header("location", location)
439        .with_header("content-type", "text/plain; charset=utf-8")
440        .with_body(b"redirect".to_vec())
441}
