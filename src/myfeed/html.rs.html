20/// Full HTML document with Bootstrap, Quill, and Clitorine.
21#[must_use]
22pub fn document(title: &str, body: &str, composer_open: bool, lang: &str) -> String {
23    let open_flag = if composer_open { "1" } else { "0" };
24    let css = asset("myfeed.css?v=5").unwrap_or_else(|_| "/assets/myfeed.css?v=5".into());
25    let js = asset("clitorine.js?v=4").unwrap_or_else(|_| "/assets/clitorine.js?v=4".into());
26    format!(
27        r#"<!DOCTYPE html>
28<html lang="{lang}">
29<head>
30<meta charset="utf-8" />
31<meta name="viewport" content="width=device-width, initial-scale=1" />
32<title>{title}</title>
33<link rel="stylesheet" href="{BOOTSTRAP_CSS}" />
34<link rel="stylesheet" href="{QUILL_CSS}" />
35<link rel="stylesheet" href="{css}" />
36</head>
37<body class="myfeed" data-composer-open="{open_flag}">
38{body}
39<script src="{BOOTSTRAP_JS}"></script>
40<script src="{QUILL_JS}"></script>
41<script src="{js}"></script>
42</body>
43</html>"#,
44        lang = escape_attr(lang),
45        title = escape_html(title),
46        css = escape_attr(&css),
47        js = escape_attr(&js),
48    )
49}

51fn locale_switcher(ui: &Ui<'_>, routes: &RouteCollection) -> String {
52    partial(|| {
53        let active = ui.locale().language();
54        let en_class = if active == "en" {
55            "btn btn-sm btn-primary"
56        } else {
57            "btn btn-sm btn-outline-secondary"
58        };
59        let fr_class = if active == "fr" {
60            "btn btn-sm btn-primary"
61        } else {
62            "btn btn-sm btn-outline-secondary"
63        };
64        let en_href =
65            path(routes, "locale_set", &[("code", "en")]).unwrap_or_else(|_| "/locale/en".into());
66        let fr_href =
67            path(routes, "locale_set", &[("code", "fr")]).unwrap_or_else(|_| "/locale/fr".into());
68        format!(
69            r#"<span class="btn-group" role="group" aria-label="Language">
70  <a class="{en_class}" href="{en_href}">{en}</a>
71  <a class="{fr_class}" href="{fr_href}">{fr}</a>
72</span>"#,
73            en_href = escape_attr(&en_href),
74            fr_href = escape_attr(&fr_href),
75            en = escape_html(&ui.t("lang_en")),
76            fr = escape_html(&ui.t("lang_fr")),
77        )
78    })
79}

309fn render_wall(view: &FeedView<'_>) -> String {
310    let posts = view.store.posts();
311    if posts.is_empty() {
312        return format!(
313            r#"<p class="text-secondary mb-0">{msg}</p>"#,
314            msg = escape_html(&view.ui.t("empty_wall")),
315        );
316    }
317    let mut posts_html = String::new();
318    for post in &posts {
319        let comments = view.store.approved_comments(post.id);
320        let comment_form = view
321            .comment_forms
322            .iter()
323            .find(|(id, _)| *id == post.id)
324            .map_or("", |(_, html)| html.as_str());
325        let like_form = view
326            .like_forms
327            .iter()
328            .find(|(id, _)| *id == post.id)
329            .map_or("", |(_, html)| html.as_str());
330        let admin_actions = view
331            .admin_forms
332            .iter()
333            .find(|(id, _)| *id == post.id)
334            .map_or("", |(_, html)| html.as_str());
335        posts_html.push_str(&post_card(
336            view.ui,
337            post,
338            &comments,
339            comment_form,
340            like_form,
341            admin_actions,
342        ));
343    }
344    posts_html
345}

404fn post_card(
405    ui: &Ui<'_>,
406    post: &Post,
407    comments: &[Comment],
408    comment_form: &str,
409    like_form: &str,
410    admin_actions: &str,
411) -> String {
412    let embed = post
413        .embed_url
414        .as_deref()
415        .and_then(embed_html)
416        .unwrap_or_default();
417    let image = post.image_data.as_deref().map_or(String::new(), |data| {
418        if data.starts_with("data:image/") && data.len() < 280_000 {
419            format!(
420                r#"<div class="embed-media my-2"><img class="img-fluid rounded" src="{src}" alt="posted image" loading="lazy" /></div>"#,
421                src = escape_attr(data)
422            )
423        } else {
424            String::new()
425        }
426    });
427    let mut comments_html = String::new();
428    for comment in comments {
429        let _ = write!(
430            comments_html,
431            r#"<p class="mb-2 small">{body}</p>"#,
432            body = escape_html(&comment.body)
433        );
434    }
435    if comments_html.is_empty() {
436        let _ = write!(
437            comments_html,
438            r#"<p class="text-secondary small mb-2">{msg}</p>"#,
439            msg = escape_html(&ui.t("no_comments"))
440        );
441    }
442    let when = if post.created_at.is_empty() {
443        String::new()
444    } else {
445        format!(" · {}", escape_html(&post.created_at))
446    };
447    format!(
448        r#"
449<article class="card post-card" id="post-{id}">
450  <div class="card-body p-4">
451    <div class="d-flex justify-content-between align-items-start gap-2 mb-2">
452      <p class="text-secondary small mb-0">Post #{id} · {category}{when}</p>
453      <div class="d-flex flex-wrap align-items-center gap-2">
454        <span class="badge text-bg-light border" data-like-count>{likes}</span>
455        {like_form}
456        {admin_actions}
457      </div>
458    </div>
459    <div class="post-body mb-2">{body}</div>
460    {image}
461    {embed}
462    <div class="border-top pt-3 mt-3">
463      <h3 class="h6">{comments_title}</h3>
464      {comments_html}
465      {comment_form}
466    </div>
467  </div>
468</article>
469"#,
470        id = post.id,
471        category = escape_html(&post.category),
472        likes = escape_html(&ui.tn("likes", i64::try_from(post.likes).unwrap_or(0))),
473        comments_title = escape_html(&ui.t("comments")),
474        body = post.body,
475    )
476}

525    }

527    let admin_href = path(routes, "admin", &[]).unwrap_or_else(|_| "/admin".into());

546      {notice_html}

663/// HTML response helper.
664#[must_use]
665pub fn html_response(status: u16, body: String) -> serenade_http::Response {
666    serenade_http::Response::new(status)
667        .with_header("content-type", "text/html; charset=utf-8")
668        .with_body(body.into_bytes())
669}
670
671/// Static asset response.
672#[must_use]
673pub fn asset_response(content_type: &str, bytes: &'static [u8]) -> serenade_http::Response {
674    serenade_http::Response::new(200)
675        .with_header("content-type", content_type)
676        .with_header("cache-control", "no-store")
677        .with_body(bytes.to_vec())
678}
679
680/// Redirect helper (paths are app-controlled).
681#[must_use]
682pub fn redirect(location: &str) -> serenade_http::Response {
683    serenade_http::Response::new(303)
684        .with_header("location", location)
685        .with_header("content-type", "text/plain; charset=utf-8")
686        .with_body(b"redirect".to_vec())
687}
688
689/// Redirect with Set-Cookie.
690#[must_use]
691pub fn redirect_with_cookie(location: &str, cookie: &str) -> serenade_http::Response {
692    redirect(location).with_header("set-cookie", cookie)
693}
