19/// Resource name used in route ids (`admin_category_*`).
20pub const CATEGORY_RESOURCE: &str = "category";

22/// Builds the categories admin resource descriptor.
23#[must_use]
24pub fn category_resource() -> AdminResource {
25    AdminResource::new(CATEGORY_RESOURCE, "/admin/categories")
26        .list_fields([AdminField::named("name")])
27        .show_fields([AdminField::named("name")])
28        .form_fields([AdminField::named("name")])
29}

31/// Registers list/show/new/edit/delete routes for the category resource.
32///
33/// # Errors
34///
35/// Propagates route name collisions.
36pub fn register_category_admin_routes(collection: &mut RouteCollection) -> Result<(), HttpError> {
37    let mut registry = AdminRegistry::new();
38    registry
39        .register(category_resource())
40        .map_err(|err| HttpError::failed(err.to_string()))?;
41    register_admin_routes(collection, &registry)
42}

57impl AdminResourceHandler for CategoryHandler {
58    fn list(&self) -> Result<Vec<AdminRow>, AdminError> {
59        Ok(self
60            .store
61            .categories()
62            .into_iter()
63            .map(|name| AdminRow::new(name.clone()).with("name", name))
64            .collect())
65    }
66
67    fn get(&self, id: &str) -> Result<Option<AdminRow>, AdminError> {
68        if self.store.has_category(id) {
69            Ok(Some(AdminRow::new(id).with("name", id)))
70        } else {
71            Ok(None)
72        }
73    }
74
75    fn create(&self, data: &HashMap<String, String>) -> Result<AdminRow, AdminError> {
76        let name = data.get("name").map_or("", String::as_str);
77        self.store
78            .add_category(name)
79            .map_err(|message| AdminError::Persist {
80                message: message.to_owned(),
81            })?;
82        let name = serenade_string::slug(name.trim());
83        let name: String = name.chars().take(40).collect();
84        Ok(AdminRow::new(name.clone()).with("name", name))
85    }
86
87    fn update(&self, id: &str, data: &HashMap<String, String>) -> Result<AdminRow, AdminError> {
88        if !self.store.has_category(id) {
89            return Err(AdminError::NotFound { id: id.to_owned() });
90        }
91        let name = serenade_string::slug(data.get("name").map_or("", String::as_str).trim());
92        if name.is_empty() {
93            return Err(AdminError::Persist {
94                message: "Category name is required.".to_owned(),
95            });
96        }
97        if name != id {
98            self.store.remove_category(id);
99            self.store
100                .add_category(&name)
101                .map_err(|message| AdminError::Persist {
102                    message: message.to_owned(),
103                })?;
104        }
105        let name: String = name.chars().take(40).collect();
106        Ok(AdminRow::new(name.clone()).with("name", name))
107    }
108
109    fn delete(&self, id: &str) -> Result<bool, AdminError> {
110        Ok(self.store.remove_category(id))
111    }
112}

133        new_href = escape_attr(&new_href),
134    ))

137/// Inputs for category admin HTTP dispatch.
138pub struct CategoryAdminCtx<'a> {
139    /// Shared feed store.
140    pub store: &'a FeedStore,
141    /// CSRF manager.
142    pub csrf: &'a HmacCsrfTokenManager,
143    /// App routes (for reverse URLs).
144    pub routes: &'a RouteCollection,
145    /// Current request.
146    pub request: &'a Request,
147    /// Localized UI strings.
148    pub ui: &'a Ui<'a>,
149    /// Whether the caller is authenticated as admin.
150    pub is_admin: bool,
151}

153/// Dispatches `admin_category_*` routes. Returns `None` when the route is unrelated.
154pub fn try_handle(
155    route: &str,
156    ctx: &CategoryAdminCtx<'_>,
157    build_login: impl FnOnce(&HmacCsrfTokenManager, &RouteCollection) -> Result<String, HttpError>,
158) -> Result<Option<Response>, HttpError> {
159    let resource = category_resource();
160    let names = [
161        resource.list_route_name(),
162        resource.show_route_name(),
163        resource.new_route_name(),
164        resource.create_route_name(),
165        resource.edit_route_name(),
166        resource.update_route_name(),
167        resource.delete_route_name(),
168    ];
169    if !names.iter().any(|name| name == route) {
170        return Ok(None);
171    }
172    if !ctx.is_admin {
173        return Ok(Some(login_denied(
174            ctx.ui,
175            ctx.routes,
176            ctx.csrf,
177            build_login,
178        )?));
179    }
180    let handler = CategoryHandler::new(ctx.store.clone());
181    match route {
182        name if name == resource.list_route_name() => {
183            Ok(Some(handle_list(&handler, &resource, ctx.routes, ctx.ui)?))
184        }
185        name if name == resource.show_route_name() => Ok(Some(handle_show(
186            &handler,
187            &resource,
188            ctx.request,
189            ctx.routes,
190            ctx.ui,
191        )?)),
192        name if name == resource.new_route_name() => {
193            Ok(Some(handle_new(&resource, ctx.csrf, ctx.routes, ctx.ui)?))
194        }
195        name if name == resource.create_route_name() => Ok(Some(handle_create(
196            &handler,
197            &resource,
198            ctx.csrf,
199            ctx.request,
200            ctx.routes,
201            ctx.ui,
202        )?)),
203        name if name == resource.edit_route_name() => Ok(Some(handle_edit(
204            &handler,
205            &resource,
206            ctx.csrf,
207            ctx.request,
208            ctx.routes,
209            ctx.ui,
210        )?)),
211        name if name == resource.update_route_name() => Ok(Some(handle_update(
212            &handler,
213            &resource,
214            ctx.csrf,
215            ctx.request,
216            ctx.routes,
217            ctx.ui,
218        )?)),
219        name if name == resource.delete_route_name() => Ok(Some(handle_delete(
220            &handler,
221            &resource,
222            ctx.csrf,
223            ctx.request,
224        )?)),
225        _ => Ok(None),
226    }
227}

229fn handle_list(

238    let body = render_list_html(resource, &rows);

250fn handle_show(
251    handler: &CategoryHandler,
252    resource: &AdminResource,
253    request: &Request,
254    routes: &RouteCollection,
255    ui: &Ui<'_>,
256) -> Result<Response, HttpError> {
257    let id = path_id(request)?;
258    let Some(row) = handler
259        .get(&id)
260        .map_err(|err| HttpError::failed(err.to_string()))?
261    else {
262        return Err(HttpError::not_found("category"));
263    };
264    let show = render_show_html(resource, &row);
265    let edit = path(routes, &resource.edit_route_name(), &[("id", id.as_str())])?;
266    let list = path(routes, &resource.list_route_name(), &[])?;
267    Ok(crud_page(
268        ui,
269        routes,
270        &format!(
271            r#"{show}
272<p class="d-flex gap-2">
273  <a class="btn btn-outline-primary btn-sm" href="{edit}">Edit</a>
274  <a class="btn btn-outline-secondary btn-sm" href="{list}">Back</a>
275</p>"#,
276            edit = escape_attr(&edit),
277            list = escape_attr(&list),
278        ),
279    ))
280}

293fn handle_create(
294    handler: &CategoryHandler,
295    resource: &AdminResource,
296    csrf: &HmacCsrfTokenManager,
297    request: &Request,
298    routes: &RouteCollection,
299    ui: &Ui<'_>,
300) -> Result<Response, HttpError> {
301    let mut form = build_resource_form(resource, resource.new_path());
302    match form.handle_request(request, csrf) {
303        Ok(FormStatus::Bound) => match handler.create(&form.data()) {
304            Ok(row) => Ok(redirect(&resource.show_path_for(row.id()))),
305            Err(err) => Ok(crud_page(
306                ui,
307                routes,
308                &format!(
309                    r#"<div class="alert alert-danger">{msg}</div>{form}"#,
310                    msg = escape_html(&err.to_string()),
311                    form = render_new_form_html(resource, csrf)
312                        .map_err(|e| HttpError::failed(e.to_string()))?
313                ),
314            )),
315        },
316        _ => Ok(redirect(&resource.new_path())),
317    }
318}

320fn handle_edit(
321    handler: &CategoryHandler,
322    resource: &AdminResource,
323    csrf: &HmacCsrfTokenManager,
324    request: &Request,
325    routes: &RouteCollection,
326    ui: &Ui<'_>,
327) -> Result<Response, HttpError> {
328    let id = path_id(request)?;
329    let Some(row) = handler
330        .get(&id)
331        .map_err(|err| HttpError::failed(err.to_string()))?
332    else {
333        return Err(HttpError::not_found("category"));
334    };
335    let edit = render_edit_form_html(resource, &row, csrf)
336        .map_err(|err| HttpError::failed(err.to_string()))?;
337    let delete = render_delete_form_html(resource, &id, csrf)
338        .map_err(|err| HttpError::failed(err.to_string()))?;
339    Ok(crud_page(ui, routes, &format!("{edit}\n{delete}")))
340}

342fn handle_update(
343    handler: &CategoryHandler,
344    resource: &AdminResource,
345    csrf: &HmacCsrfTokenManager,
346    request: &Request,
347    routes: &RouteCollection,
348    ui: &Ui<'_>,
349) -> Result<Response, HttpError> {
350    let id = path_id(request)?;
351    let mut form = build_resource_form(resource, resource.edit_path_for(&id));
352    match form.handle_request(request, csrf) {
353        Ok(FormStatus::Bound) => match handler.update(&id, &form.data()) {
354            Ok(row) => Ok(redirect(&resource.show_path_for(row.id()))),
355            Err(err) => {
356                let Some(row) = handler
357                    .get(&id)
358                    .map_err(|e| HttpError::failed(e.to_string()))?
359                else {
360                    return Err(HttpError::not_found("category"));
361                };
362                let edit = render_edit_form_html(resource, &row, csrf)
363                    .map_err(|e| HttpError::failed(e.to_string()))?;
364                Ok(crud_page(
365                    ui,
366                    routes,
367                    &format!(
368                        r#"<div class="alert alert-danger">{msg}</div>{edit}"#,
369                        msg = escape_html(&err.to_string()),
370                    ),
371                ))
372            }
373        },
374        _ => Ok(redirect(&resource.edit_path_for(&id))),
375    }
376}

378fn handle_delete(
379    handler: &CategoryHandler,
380    resource: &AdminResource,
381    csrf: &HmacCsrfTokenManager,
382    request: &Request,
383) -> Result<Response, HttpError> {
384    let id = path_id(request)?;
385    let mut form = Form::builder(format!("admin_{}_delete", resource.name()))
386        .action(resource.delete_path_for(&id))
387        .build();
388    if form.handle_request(request, csrf).is_err() {
389        return Ok(redirect(&resource.edit_path_for(&id)));
390    }
391    let _ = handler
392        .delete(&id)
393        .map_err(|err| HttpError::failed(err.to_string()))?;
394    Ok(redirect(&resource.list_path()))
395}

397fn path_id(request: &Request) -> Result<String, HttpError> {
398    request
399        .attributes()
400        .get::<String>("id")
401        .cloned()
402        .ok_or_else(|| HttpError::bad_request("missing id"))
403}

405fn crud_page(ui: &Ui<'_>, routes: &RouteCollection, inner: &str) -> Response {
406    let feed = path(routes, "feed", &[]).unwrap_or_else(|_| "/".into());
407    let admin = path(routes, "admin", &[]).unwrap_or_else(|_| "/admin".into());
408    let body = format!(
409        r#"
410<div class="myfeed-shell">
411  <header class="d-flex flex-wrap gap-3 mb-4 pb-3 border-bottom">
412    <div class="me-auto">
413      <h1 class="myfeed-brand">{title}</h1>
414      <p class="myfeed-tag">serenade-admin categories</p>
415    </div>
416    <nav class="myfeed-nav d-flex align-items-center gap-3 pt-2">
417      <a class="link-secondary" href="{feed}">{nav_feed}</a>
418      <a class="admin-link btn btn-outline-primary btn-sm" href="{admin}">{nav_admin}</a>
419    </nav>
420  </header>
421  <main>{inner}</main>
422</div>
423"#,
424        title = escape_html(&ui.t("categories_title")),
425        feed = escape_attr(&feed),
426        admin = escape_attr(&admin),
427        nav_feed = escape_html(&ui.t("nav_feed")),
428        nav_admin = escape_html(&ui.t("nav_admin")),
429    );
430    html_response(
431        200,
432        document(
433            &ui.t("categories_title"),
434            &body,
435            false,
436            ui.locale().language(),
437        ),
438    )
439}

441/// Login denial helper shared with main admin handlers.
442pub fn login_denied(
443    ui: &Ui<'_>,
444    routes: &RouteCollection,
445    csrf: &HmacCsrfTokenManager,
446    build_login: impl FnOnce(&HmacCsrfTokenManager, &RouteCollection) -> Result<String, HttpError>,
447) -> Result<Response, HttpError> {
448    let login = build_login(csrf, routes)?;
449    Ok(html_response(
450        401,
451        admin_login_page(ui, routes, &login, Some("Sign in first.")),
452    ))
453}

460    #[test]
461    fn category_handler_crud_roundtrip() {
462        let store = FeedStore::open_memory();
463        let handler = CategoryHandler::new(store);
464        let created = handler
465            .create(&HashMap::from([("name".to_owned(), "Dogs".to_owned())]))
466            .expect("create");
467        assert_eq!(created.id(), "dogs");
468        assert!(handler.get("dogs").expect("get").is_some());
469        let updated = handler
470            .update(
471                "dogs",
472                &HashMap::from([("name".to_owned(), "Cats".to_owned())]),
473            )
474            .expect("update");
475        assert_eq!(updated.id(), "cats");
476        assert!(handler.delete("cats").expect("delete"));
477        assert!(handler.get("cats").expect("missing").is_none());
478    }
