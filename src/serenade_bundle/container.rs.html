1//! Assemble a DI container from package config and extensions.
2
3use std::path::Path;
4
5use serenade_config::{load_packages_for_env, Config};
6use serenade_console::RegisterCommandsPass;
7use serenade_di::{Container, ContainerBuilder, ServiceDefinition};
8use serenade_event::RegisterEventSubscribersPass;
9
10use crate::framework::CONFIG_SERVICE;
11use crate::{BundleError, Extension};
12
13/// Loads `config/packages` (plus `{environment}/` overlay), runs `extensions`, and compiles.
14///
15/// Flattened package parameters are applied first. The merged root config is
16/// registered as [`CONFIG_SERVICE`]. Each extension then receives
17/// `config.section(extension.alias())`. Event and console compile passes always
18/// run so tagged subscribers and commands populate their services.
19///
20/// Call [`serenade_config::load_dotenv`] on the project root before this when
21/// apps use `.env` files.
22///
23/// # Errors
24///
25/// Returns [`BundleError`] on config, extension, or DI failures.
26pub fn build_container(
27    packages_dir: Option<&Path>,
28    environment: &str,
29    extensions: &[&dyn Extension],
30) -> Result<(Config, Container), BundleError> {
31    let config = match packages_dir {
32        Some(dir) => load_packages_for_env(dir, environment)?.interpolate_env()?,
33        None => Config::empty(),
34    };
35    let mut builder = ContainerBuilder::new();
36    config.apply_to(builder.parameters_mut());
37    let snapshot = config.clone();
38    builder.register(ServiceDefinition::new(CONFIG_SERVICE), move |_| {
39        Ok(Box::new(snapshot.clone()))
40    })?;
41    builder.add_compile_pass(RegisterEventSubscribersPass);
42    builder.add_compile_pass(RegisterCommandsPass);
43    for extension in extensions {
44        let section = config.section(extension.alias());
45        extension.load(&section, &mut builder).map_err(|error| {
46            if matches!(error, BundleError::Extension { .. }) {
47                error
48            } else {
49                BundleError::Extension {
50                    alias: extension.alias(),
51                    message: error.to_string(),
52                }
53            }
54        })?;
55    }
56    Ok((config, builder.compile()?))
57}
