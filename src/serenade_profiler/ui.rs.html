12/// Handles `/_profiler` and `/_profiler/{token}` when the path matches `prefix`.
13///
14/// Returns `None` when the request is not a profiler UI route.
15#[must_use]
16pub fn try_handle_profiler(
17    store: &Arc<ProfileStore>,
18    prefix: &str,
19    request: &Request,
20) -> Option<Response> {
21    if request.method() != Method::Get {
22        return None;
23    }
24    let path = request.path();
25    let prefix = prefix.trim_end_matches('/');
26    if path == prefix || path == format!("{prefix}/") {
27        return Some(
28            Response::new(200)
29                .with_header("content-type", "text/html; charset=utf-8")
30                .with_body(profiler_index_html(store, prefix).into_bytes()),
31        );
32    }
33    let token_prefix = format!("{prefix}/");
34    if let Some(token) = path.strip_prefix(&token_prefix) {
35        if token.contains('/') {
36            return None;
37        }
38        return Some(store.get(token).map_or_else(
39            || Response::text(404, format!("unknown profiler token `{token}`")),
40            |profile| {
41                Response::new(200)
42                    .with_header("content-type", "text/html; charset=utf-8")
43                    .with_body(profiler_detail_html(&profile, prefix).into_bytes())
44            },
45        ));
46    }
47    None
48}

50/// Index page listing recent profiles.
51#[must_use]
52pub fn profiler_index_html(store: &ProfileStore, prefix: &str) -> String {
53    let prefix = prefix.trim_end_matches('/');
54    let mut rows = String::new();
55    for profile in store.list() {
56        rows.push_str(&format!(
57            r#"<tr><td><a href="{prefix}/{token}">{token}</a></td><td>{method}</td><td>{path}</td><td>{status}</td><td>{ms} ms</td></tr>"#,
58            prefix = escape(prefix),
59            token = escape(&profile.token),
60            method = escape(&profile.method),
61            path = escape(&profile.path),
62            status = profile.status,
63            ms = profile.duration.as_millis(),
64        ));
65    }
66    if rows.is_empty() {
67        rows.push_str(r#"<tr><td colspan="5">No profiles yet.</td></tr>"#);
68    }
69    format!(
70        r#"<!DOCTYPE html><html><head><meta charset="utf-8"><title>Serenade profiler</title>
71<style>body{{font:14px/1.4 system-ui,sans-serif;margin:1.5rem;}}table{{border-collapse:collapse;width:100%;}}td,th{{border:1px solid #ccc;padding:6px 8px;text-align:left;}}</style>
72</head><body><h1>Profiler</h1><p>Recent requests (newest first).</p>
73<table><thead><tr><th>Token</th><th>Method</th><th>Path</th><th>Status</th><th>Time</th></tr></thead><tbody>{rows}</tbody></table>
74<p><a href="/">Back to app</a></p></body></html>"#
75    )
76}

78/// Detail page for one profile.
79#[must_use]
80pub fn profiler_detail_html(profile: &ProfileData, prefix: &str) -> String {
81    let prefix = prefix.trim_end_matches('/');
82    let route = profile
83        .route
84        .as_deref()
85        .map_or_else(|| "—".to_owned(), escape);
86    let mut queries = String::new();
87    if profile.queries.is_empty() {
88        queries.push_str("<li>No queries recorded.</li>");
89    } else {
90        for query in &profile.queries {
91            queries.push_str(&format!(
92                "<li><code>{}</code> · {} ms{}</li>",
93                escape(&query.sql),
94                query.duration.as_millis(),
95                query
96                    .binds_summary
97                    .as_ref()
98                    .map(|b| format!(" · binds: {}", escape(b)))
99                    .unwrap_or_default()
100            ));
101        }
102    }
103    let mut logs = String::new();
104    if profile.logs.is_empty() {
105        logs.push_str("<li>No log lines captured.</li>");
106    } else {
107        for line in &profile.logs {
108            logs.push_str(&format!(
109                "<li><strong>{}</strong> [{}] {}</li>",
110                escape(&line.level),
111                escape(&line.target),
112                escape(&line.message)
113            ));
114        }
115    }
116    format!(
117        r#"<!DOCTYPE html><html><head><meta charset="utf-8"><title>Profile {token}</title>
118<style>body{{font:14px/1.4 system-ui,sans-serif;margin:1.5rem;max-width:960px;}}code{{background:#f4f4f4;padding:1px 4px;}}section{{margin:1.25rem 0;}}h2{{margin-bottom:0.4rem;}}</style>
119</head><body>
120<p><a href="{prefix}">All profiles</a></p>
121<h1>Profile {token}</h1>
122<section><h2>Request</h2>
123<ul>
124<li>Method: <code>{method}</code></li>
125<li>Path: <code>{path}</code></li>
126<li>Status: {status}</li>
127<li>Duration: {ms} ms</li>
128<li>Route: <code>{route}</code></li>
129</ul></section>
130<section><h2>Timing</h2><p>{ms} ms wall time inside profiler middleware.</p></section>
131<section><h2>Logs</h2><ul>{logs}</ul></section>
132<section><h2>Database</h2><ul>{queries}</ul></section>
133<section><h2>Views</h2><p>No view collector yet.</p></section>
134</body></html>"#,
135        prefix = escape(prefix),
136        token = escape(&profile.token),
137        method = escape(&profile.method),
138        path = escape(&profile.path),
139        status = profile.status,
140        ms = profile.duration.as_millis(),
141        route = route,
142        logs = logs,
143        queries = queries,
144    )
145}

147fn escape(input: &str) -> String {
148    let mut out = String::with_capacity(input.len());
149    for ch in input.chars() {
150        match ch {
151            '&' => out.push_str("&amp;"),
152            '<' => out.push_str("&lt;"),
153            '>' => out.push_str("&gt;"),
154            '"' => out.push_str("&quot;"),
155            '\'' => out.push_str("&#39;"),
156            c => out.push(c),
157        }
158    }
159    out
160}
