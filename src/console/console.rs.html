1//! Sample console entry for the demo app (`bin/console` analogue).
2
3use std::path::PathBuf;
4use std::process::ExitCode;
5use std::sync::Arc;
6
7use serenade_bundle::{
8    build_container, FrameworkBundle, FrameworkExtension, CONSOLE_APPLICATION_SERVICE,
9};
10use serenade_console::Application;
11use serenade_kernel::{App, Application as KernelApp, Environment};
12
13fn main() -> ExitCode {
14    match run() {
15        Ok(()) => ExitCode::SUCCESS,
16        Err(error) => {
17            eprintln!("{error}");
18            ExitCode::FAILURE
19        }
20    }
21}
22
23fn run() -> Result<(), Box<dyn std::error::Error>> {
24    let env_name = std::env::var("APP_ENV").unwrap_or_else(|_| "dev".to_owned());
25    let environment = Environment::from_name(&env_name)?;
26    let root = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
27    serenade_config::load_dotenv(&root, environment.as_str())?;
28
29    let mut app = App::new(environment.clone());
30    app.register_bundle(FrameworkBundle)?;
31    app.boot()?;
32
33    let packages = root.join("config/packages");
34    let (_config, container) = build_container(
35        Some(packages.as_path()),
36        environment.as_str(),
37        &[&FrameworkExtension],
38    )?;
39    let container = Arc::new(container);
40    let console = container.get_as::<Application>(CONSOLE_APPLICATION_SERVICE)?;
41    let argv: Vec<String> = std::env::args().collect();
42    console.run_with(argv, Some(Arc::clone(&container)))?;
43    app.shutdown()?;
44    Ok(())
45}
