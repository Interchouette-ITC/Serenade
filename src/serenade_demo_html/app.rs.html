143fn admin_path(value: &str) -> Result<String, String> {
144    let segment = value.strip_prefix('/').unwrap_or(value);
145    if segment.is_empty()
146        || !segment
147            .bytes()
148            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-' || byte == b'_')
149        || ["feed", "posts", "search", "assets", "locale", "_profiler"].contains(&segment)
150    {
151        return Err("MYFEED_ADMIN_PATH must be a non-reserved single URL segment containing only letters, digits, '-' or '_'".into());
152    }
153    Ok(format!("/{segment}"))
154}

156fn handle(state: &AppState, request: &mut Request) -> Result<Response, HttpError> {
157    if let Some(response) = try_handle_profiler(&state.profiler, "/_profiler", request) {
158        return Ok(response);
159    }
160    state.matcher.apply(request)?;
161    let _locale = state.negotiator.apply(request);
162    tracing::info!(target: REQUEST, path = request.path(), "myfeed request");
163    let route = request
164        .attributes()
165        .get::<String>(ROUTE_ATTRIBUTE)
166        .map_or("", String::as_str);
167
168    match route {
169        "feed" => feed_with_flash(state, request, None, false, false),
170        "feed_latest" => handle_latest(state),
171        "post_show" | "comments_show" => handle_post_get(state, request),
172        "post_image" => handle_post_image(state, request),
173        "search" => Ok(handle_search(state, request)),
174        "post_create" => handle_post_create(state, request),
175        "comment_create" => handle_comment_create(state, request),
176        "post_like" => handle_like(state, request),
177        "admin" => handle_admin_get(state, request),
178        "admin_login" => handle_admin_login(state, request),
179        "admin_logout" => Ok(handle_admin_logout(state, request)),
180        "admin_approve" => handle_admin_moderation(state, request, true),
181        "admin_reject" => handle_admin_moderation(state, request, false),
182        "admin_post_edit_get" => handle_post_edit_get(state, request),
183        "admin_post_edit_post" => handle_post_edit_post(state, request),
184        "admin_post_delete" => handle_post_delete(state, request),
185        "asset_css" => Ok(theme_css_response(state)),
186        "asset_theme" => theme_asset_response(state, request),
187        "asset_js" => Ok(asset_response(
188            "text/javascript; charset=utf-8",
189            include_bytes!("../assets/clitorine.min.js"),
190        )),
191        "asset_favicon" => Ok(asset_response(
192            "image/png",
193            include_bytes!("../assets/favicon.png"),
194        )),
195        "asset_favicon_ico" => Ok(favicon_response(state)),
196        "locale_set" => Ok(handle_locale_set(state, request)),
197        _ => {
198            let ui = ui_for(state, request);
199            admin_crud::try_handle(
200                route,
201                &admin_crud::CategoryAdminCtx {
202                    store: &state.store,
203                    csrf: &state.csrf,
204                    routes: state.matcher.collection(),
205                    request,
206                    ui: &ui,
207                    is_admin: is_admin(request, &state.admin_token),
208                },
209                build_login_form,
210            )?
211            .map_or_else(|| Err(HttpError::not_found("no handler")), Ok)
212        }
213    }
214}

216fn favicon_response(state: &AppState) -> Response {
217    let Some(path) = &state.favicon_file else {
218        return asset_response("image/x-icon", include_bytes!("../assets/favicon.ico"));
219    };
220    match std::fs::read(path) {
221        Ok(bytes) => Response::new(200)
222            .with_header("content-type", "image/x-icon")
223            .with_header("cache-control", "no-store")
224            .with_body(bytes),
225        Err(error) => {
226            tracing::error!(path = %path.display(), %error, "MyFeed favicon could not be read");
227            Response::text(500, "MyFeed favicon could not be read.")
228                .with_header("cache-control", "no-store")
229        }
230    }
231}

233fn theme_asset_response(state: &AppState, request: &Request) -> Result<Response, HttpError> {
234    let not_found = || HttpError::not_found("theme asset not found");
235    let name = request
236        .attributes()
237        .get::<String>("name")
238        .ok_or_else(not_found)?;
239    let (stem, extension) = name.rsplit_once('.').ok_or_else(not_found)?;
240    if stem.is_empty()
241        || !stem
242            .bytes()
243            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-' || byte == b'_')
244    {
245        return Err(not_found());
246    }
247    let content_type = match extension {
248        "png" => "image/png",
249        "jpg" | "jpeg" => "image/jpeg",
250        "webp" => "image/webp",
251        "gif" => "image/gif",
252        "js" => "text/javascript; charset=utf-8",
253        _ => return Err(not_found()),
254    };
255    let path = state
256        .theme_css
257        .as_ref()
258        .ok_or_else(not_found)?
259        .with_file_name(name);
260    match std::fs::symlink_metadata(&path) {
261        Ok(metadata) if metadata.file_type().is_file() => {}
262        Ok(_) => return Err(not_found()),
263        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Err(not_found()),
264        Err(error) => {
265            tracing::error!(path = %path.display(), %error, "theme asset could not be inspected");
266            return Ok(Response::text(500, "Theme asset could not be read.")
267                .with_header("cache-control", "no-store"));
268        }
269    }
270    match std::fs::read(&path) {
271        Ok(bytes) => Ok(Response::new(200)
272            .with_header("content-type", content_type)
273            .with_header("cache-control", "no-store")
274            .with_body(bytes)),
275        Err(error) => {
276            tracing::error!(path = %path.display(), %error, "theme asset could not be read");
277            Ok(Response::text(500, "Theme asset could not be read.")
278                .with_header("cache-control", "no-store"))
279        }
280    }
281}

283fn theme_css_response(state: &AppState) -> Response {
284    let mut css = include_bytes!("../assets/myfeed.css").to_vec();
285    if let Some(path) = &state.theme_css {
286        match std::fs::read(path) {
287            Ok(theme) => {
288                css.push(b'\n');
289                css.extend_from_slice(&theme);
290            }
291            Err(error) => {
292                tracing::error!(
293                    path = %path.display(),
294                    error = %error,
295                    "configured MyFeed theme CSS could not be read"
296                );
297                return Response::text(500, "Configured MyFeed theme CSS could not be read.")
298                    .with_header("cache-control", "no-store");
299            }
300        }
301    }
302
303    Response::new(200)
304        .with_header("content-type", "text/css; charset=utf-8")
305        .with_header("cache-control", "no-store")
306        .with_body(css)
307}

330fn validate_theme_files(
331    theme: Option<&Path>,
332) -> Result<Option<PathBuf>, Box<dyn std::error::Error>> {
333    let Some(theme) = theme else {
334        return Ok(None);
335    };
336    std::fs::read(theme).map_err(|error| {
337        format!(
338            "MYFEED_THEME_CSS={} could not be read: {error}",
339            theme.display()
340        )
341    })?;
342    let path = theme.with_file_name("favicon.ico");
343    if path
344        .try_exists()
345        .map_err(|error| format!("check favicon {}: {error}", path.display()))?
346    {
347        std::fs::read(&path)
348            .map_err(|error| format!("read favicon {}: {error}", path.display()))?;
349        Ok(Some(path))
350    } else {
351        Ok(None)
352    }
353}

355pub fn run() -> Result<(), Box<dyn std::error::Error>> {
356    let bind = std::env::var("MYFEED_BIND").unwrap_or_else(|_| DEFAULT_BIND.to_owned());
357    let csrf_secret =
358        std::env::var("MYFEED_CSRF_SECRET").unwrap_or_else(|_| DEFAULT_CSRF.to_owned());
359    let admin_token =
360        std::env::var("MYFEED_ADMIN_TOKEN").unwrap_or_else(|_| DEFAULT_ADMIN.to_owned());
361    let admin_path = match std::env::var("MYFEED_ADMIN_PATH") {
362        Ok(value) => admin_path(&value)?,
363        Err(std::env::VarError::NotPresent) => "/admin".to_owned(),
364        Err(error) => return Err(format!("invalid MYFEED_ADMIN_PATH: {error}").into()),
365    };
366    let theme_css = std::env::var_os("MYFEED_THEME_CSS").map(PathBuf::from);
367    let footer_html = std::env::var_os("MYFEED_FOOTER_FILE")
368        .map(PathBuf::from)
369        .map(|path| {
370            std::fs::read_to_string(&path).map_err(|error| {
371                format!(
372                    "MYFEED_FOOTER_FILE={} could not be read: {error}",
373                    path.display()
374                )
375            })
376        })
377        .transpose()?;
378    let favicon_file = validate_theme_files(theme_css.as_deref())?;
379
380    let db_path = std::env::var("MYFEED_DB").unwrap_or_else(|_| ".myfeed.sqlite".to_owned());
381    let store = FeedStore::open(db_path);
382    seed_if_empty(&store);
383    let index = MemorySearchAdapter::new();
384    rebuild_search_index(&store, &index)?;
385    let translations = PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("translations");
386    if !is_dir(&translations) {
387        return Err(format!("translations dir missing: {}", translations.display()).into());
388    }
389    let catalogs = Finder::new()
390        .in_path(&translations)
391        .files()
392        .name("*.toml")
393        .collect()
394        .map_err(|err| format!("find translations: {err}"))?;
395    if catalogs.is_empty() {
396        return Err("no translation toml files found".into());
397    }
398    let mut translator = Translator::new(Locale::new("en").expect("en"))
399        .with_fallbacks(vec![Locale::new("en").expect("en")]);
400    translator
401        .load_path(&translations)
402        .map_err(|err| format!("load translations: {err}"))?;
403    let negotiator = LocaleNegotiator::new(Locale::new("en").expect("en"))
404        .with_allowed(vec![
405            Locale::new("en").expect("en"),
406            Locale::new("fr").expect("fr"),
407        ])
408        .with_cookie_name(LOCALE_COOKIE);
409
410    let profiler_enabled = std::env::var("MYFEED_PROFILER").map_or(true, |value| {
411        value != "0" && !value.eq_ignore_ascii_case("false")
412    });
413    let profiler_store = Arc::new(ProfileStore::new(50));
414    let _ = tracing_subscriber::registry()
415        .with(ProfilerLogLayer)
416        .with(tracing_subscriber::fmt::layer().with_target(true))
417        .try_init();
418
419    let state = Arc::new(AppState {
420        store,
421        index,
422        csrf: HmacCsrfTokenManager::new(csrf_secret.as_bytes()),
423        matcher: UrlMatcher::new(routes(&admin_path)?),
424        admin_token: admin_token.clone(),
425        translator,
426        negotiator,
427        profiler: Arc::clone(&profiler_store),
428        notifier: NullTransport::new(),
429        theme_css,
430        footer_html,
431        show_admin_link: admin_path == "/admin",
432        favicon_file,
433    });
434
435    let state_for_handler = Arc::clone(&state);
436    let mut async_kernel = AsyncHttpKernel::from_sync(move |request: &mut Request| {
437        handle(state_for_handler.as_ref(), request)
438    });
439    async_kernel.push_middleware(serenade_minify::HtmlMinifier);
440    if profiler_enabled {
441        async_kernel.push_middleware(AsyncProfilerMiddleware::new(
442            Arc::clone(&profiler_store),
443            ProfilerConfig::enabled(50),
444        ));
445    }
446
447    println!("demo-html listening on http://{bind}/");
448    println!("Admin: open {admin_path} and sign in with token `{admin_token}`");
449    if profiler_enabled {
450        println!("Profiler: http://{bind}/_profiler (disable with MYFEED_PROFILER=0)");
451    }
452
453    let runtime = tokio::runtime::Runtime::new()?;
454    runtime.block_on(serenade_http_actix::listen(bind, async_kernel))?;
455    Ok(())
456}
