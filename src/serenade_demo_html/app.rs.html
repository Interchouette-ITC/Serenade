144fn admin_path(value: &str) -> Result<String, String> {
145    let segment = value.strip_prefix('/').unwrap_or(value);
146    if segment.is_empty()
147        || !segment
148            .bytes()
149            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-' || byte == b'_')
150        || ["feed", "posts", "search", "assets", "locale", "_profiler"].contains(&segment)
151    {
152        return Err("MYFEED_ADMIN_PATH must be a non-reserved single URL segment containing only letters, digits, '-' or '_'".into());
153    }
154    Ok(format!("/{segment}"))
155}

157fn handle(state: &AppState, request: &mut Request) -> Result<Response, HttpError> {
158    if let Some(response) = try_handle_profiler(&state.profiler, "/_profiler", request) {
159        return Ok(response);
160    }
161    state.matcher.apply(request)?;
162    let _locale = state.negotiator.apply(request);
163    tracing::info!(target: REQUEST, path = request.path(), "myfeed request");
164    let route = request
165        .attributes()
166        .get::<String>(ROUTE_ATTRIBUTE)
167        .map_or("", String::as_str);
168
169    match route {
170        "feed" => feed_with_flash(state, request, None, false, false)
171            .map(|response| preload_theme_image(state, request, response)),
172        "feed_latest" => handle_latest(state),
173        "post_show" | "comments_show" => handle_post_get(state, request),
174        "post_image" => handle_post_image(state, request),
175        "search" => Ok(preload_theme_image(
176            state,
177            request,
178            handle_search(state, request),
179        )),
180        "post_create" => handle_post_create(state, request),
181        "comment_create" => handle_comment_create(state, request),
182        "post_like" => handle_like(state, request),
183        "admin" => handle_admin_get(state, request),
184        "admin_login" => handle_admin_login(state, request),
185        "admin_logout" => Ok(handle_admin_logout(state, request)),
186        "admin_approve" => handle_admin_moderation(state, request, true),
187        "admin_reject" => handle_admin_moderation(state, request, false),
188        "admin_post_edit_get" => handle_post_edit_get(state, request),
189        "admin_post_edit_post" => handle_post_edit_post(state, request),
190        "admin_post_delete" => handle_post_delete(state, request),
191        "asset_css" => Ok(theme_css_response(state)),
192        "asset_theme" => theme_asset_response(state, request),
193        "asset_js" => Ok(asset_response(
194            "text/javascript; charset=utf-8",
195            include_bytes!("../assets/clitorine.min.js"),
196        )),
197        "asset_favicon" => Ok(asset_response(
198            "image/png",
199            include_bytes!("../assets/favicon.png"),
200        )),
201        "asset_favicon_ico" => Ok(favicon_response(state)),
202        "locale_set" => Ok(handle_locale_set(state, request)),
203        _ => {
204            let ui = ui_for(state, request);
205            admin_crud::try_handle(
206                route,
207                &admin_crud::CategoryAdminCtx {
208                    store: &state.store,
209                    csrf: &state.csrf,
210                    routes: state.matcher.collection(),
211                    request,
212                    ui: &ui,
213                    is_admin: is_admin(request, &state.admin_token),
214                },
215                build_login_form,
216            )?
217            .map_or_else(|| Err(HttpError::not_found("no handler")), Ok)
218        }
219    }
220}

222fn preload_theme_image(state: &AppState, request: &Request, response: Response) -> Response {

225    }

230        ),
231        None => response,
232    }

235fn theme_asset_type(name: &str) -> Option<&'static str> {
236    let (stem, extension) = name.rsplit_once('.')?;
237    if stem.is_empty()
238        || !stem
239            .bytes()
240            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-' || byte == b'_')
241    {
242        return None;
243    }
244    match extension {
245        "png" => Some("image/png"),
246        "jpg" | "jpeg" => Some("image/jpeg"),
247        "webp" => Some("image/webp"),
248        "gif" => Some("image/gif"),
249        "js" => Some("text/javascript; charset=utf-8"),
250        _ => None,
251    }
252}

254fn favicon_response(state: &AppState) -> Response {

256        return asset_response("image/x-icon", include_bytes!("../assets/favicon.ico"));
257    };
258    match std::fs::read(path) {

260            .with_header("content-type", "image/x-icon")
261            .with_header("cache-control", "no-store")
262            .with_body(bytes),
263        Err(error) => {
264            tracing::error!(path = %path.display(), %error, "MyFeed favicon could not be read");
265            Response::text(500, "MyFeed favicon could not be read.")
266                .with_header("cache-control", "no-store")
267        }
268    }

271fn theme_asset_response(state: &AppState, request: &Request) -> Result<Response, HttpError> {
272    let not_found = || HttpError::not_found("theme asset not found");
273    let name = request
274        .attributes()
275        .get::<String>("name")
276        .ok_or_else(not_found)?;
277    let content_type = theme_asset_type(name).ok_or_else(not_found)?;
278    let path = state
279        .theme_css
280        .as_ref()
281        .ok_or_else(not_found)?
282        .with_file_name(name);
283    match std::fs::symlink_metadata(&path) {
284        Ok(metadata) if metadata.file_type().is_file() => {}
285        Ok(_) => return Err(not_found()),
286        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Err(not_found()),
287        Err(error) => {
288            tracing::error!(path = %path.display(), %error, "theme asset could not be inspected");
289            return Ok(Response::text(500, "Theme asset could not be read.")
290                .with_header("cache-control", "no-store"));
291        }
292    }
293    match std::fs::read(&path) {
294        Ok(bytes) if content_type.starts_with("image/") => {
295            let mut hash = std::collections::hash_map::DefaultHasher::new();
296            bytes.hash(&mut hash);
297            content_type.hash(&mut hash);
298            let etag = format!("\"{:x}\"", hash.finish());
299            let response = if request.headers().get("if-none-match") == Some(etag.as_str()) {
300                Response::new(304)
301            } else {
302                Response::new(200)
303                    .with_header("content-type", content_type)
304                    .with_body(bytes)
305            };
306            Ok(response
307                .with_header("etag", etag)
308                .with_header("cache-control", "no-cache"))
309        }
310        Ok(bytes) => Ok(Response::new(200)
311            .with_header("content-type", content_type)
312            .with_header("cache-control", "no-store")
313            .with_body(bytes)),
314        Err(error) => {
315            tracing::error!(path = %path.display(), %error, "theme asset could not be read");
316            Ok(Response::text(500, "Theme asset could not be read.")
317                .with_header("cache-control", "no-store"))
318        }
319    }
320}

322fn theme_css_response(state: &AppState) -> Response {
323    let mut css = include_bytes!("../assets/myfeed.css").to_vec();
324    if let Some(path) = &state.theme_css {
325        match std::fs::read(path) {
326            Ok(theme) => {
327                css.push(b'\n');
328                css.extend_from_slice(&theme);
329            }
330            Err(error) => {
331                tracing::error!(
332                    path = %path.display(),
333                    error = %error,
334                    "configured MyFeed theme CSS could not be read"
335                );
336                return Response::text(500, "Configured MyFeed theme CSS could not be read.")
337                    .with_header("cache-control", "no-store");
338            }
339        }
340    }
341
342    Response::new(200)
343        .with_header("content-type", "text/css; charset=utf-8")
344        .with_header("cache-control", "no-store")
345        .with_body(css)
346}

369fn validate_theme_files(
370    theme: Option<&Path>,

376        format!(

382    if path
383        .try_exists()
384        .map_err(|error| format!("check favicon {}: {error}", path.display()))?
385    {
386        std::fs::read(&path)
387            .map_err(|error| format!("read favicon {}: {error}", path.display()))?;
388        Ok(Some(path))
389    } else {
390        Ok(None)
391    }

394fn theme_preload_image(
395    theme_css: Option<&Path>,
396    configured: Option<String>,
397) -> Result<Option<String>, Box<dyn std::error::Error>> {
398    if let Some(name) = configured {
399        if !theme_asset_type(&name).is_some_and(|kind| kind.starts_with("image/"))
400            || theme_css.is_none()
401        {
402            return Err("MYFEED_THEME_PRELOAD_IMAGE must name a supported theme image and requires MYFEED_THEME_CSS".into());
403        }
404        return Ok(Some(name));
405    }
406
407    let Some(theme_css) = theme_css else {
408        return Ok(None);
409    };
410    let path = theme_css.with_file_name("header.webp");
411    match std::fs::symlink_metadata(&path) {
412        Ok(metadata) if metadata.file_type().is_file() => Ok(Some("header.webp".into())),
413        Ok(_) => Ok(None),
414        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(None),
415        Err(error) => Err(format!("check theme header {}: {error}", path.display()).into()),
416    }
417}

419fn load_translator() -> Result<Translator, Box<dyn std::error::Error>> {
420    let translations = PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("translations");
421    if !is_dir(&translations) {
422        return Err(format!("translations dir missing: {}", translations.display()).into());
423    }
424    let catalogs = Finder::new()
425        .in_path(&translations)
426        .files()
427        .name("*.toml")
428        .collect()
429        .map_err(|err| format!("find translations: {err}"))?;
430    if catalogs.is_empty() {
431        return Err("no translation toml files found".into());
432    }
433    let mut translator = Translator::new(Locale::new("en").expect("en"))
434        .with_fallbacks(vec![Locale::new("en").expect("en")]);
435    translator
436        .load_path(&translations)
437        .map_err(|err| format!("load translations: {err}"))?;
438    Ok(translator)
439}

441pub fn run() -> Result<(), Box<dyn std::error::Error>> {
442    let bind = std::env::var("MYFEED_BIND").unwrap_or_else(|_| DEFAULT_BIND.to_owned());
443    let csrf_secret =
444        std::env::var("MYFEED_CSRF_SECRET").unwrap_or_else(|_| DEFAULT_CSRF.to_owned());
445    let admin_token =
446        std::env::var("MYFEED_ADMIN_TOKEN").unwrap_or_else(|_| DEFAULT_ADMIN.to_owned());
447    let admin_path = match std::env::var("MYFEED_ADMIN_PATH") {
448        Ok(value) => admin_path(&value)?,
449        Err(std::env::VarError::NotPresent) => "/admin".to_owned(),
450        Err(error) => return Err(format!("invalid MYFEED_ADMIN_PATH: {error}").into()),
451    };
452    let theme_css = std::env::var_os("MYFEED_THEME_CSS").map(PathBuf::from);
453    let configured_preload_image = match std::env::var("MYFEED_THEME_PRELOAD_IMAGE") {
454        Ok(name) => Some(name),
455        Err(std::env::VarError::NotPresent) => None,
456        Err(error) => return Err(format!("invalid MYFEED_THEME_PRELOAD_IMAGE: {error}").into()),
457    };
458    let theme_preload_image = theme_preload_image(theme_css.as_deref(), configured_preload_image)?;
459    let footer_html = std::env::var_os("MYFEED_FOOTER_FILE")
460        .map(PathBuf::from)
461        .map(|path| {
462            std::fs::read_to_string(&path).map_err(|error| {
463                format!(
464                    "MYFEED_FOOTER_FILE={} could not be read: {error}",
465                    path.display()
466                )
467            })
468        })
469        .transpose()?;
470    let favicon_file = validate_theme_files(theme_css.as_deref())?;
471
472    let db_path = std::env::var("MYFEED_DB").unwrap_or_else(|_| ".myfeed.sqlite".to_owned());
473    let store = FeedStore::open(db_path);
474    seed_if_empty(&store);
475    let index = MemorySearchAdapter::new();
476    rebuild_search_index(&store, &index)?;
477    let translator = load_translator()?;
478    let negotiator = LocaleNegotiator::new(Locale::new("en").expect("en"))
479        .with_allowed(vec![
480            Locale::new("en").expect("en"),
481            Locale::new("fr").expect("fr"),
482        ])
483        .with_cookie_name(LOCALE_COOKIE);
484
485    let profiler_enabled = std::env::var("MYFEED_PROFILER").map_or(true, |value| {
486        value != "0" && !value.eq_ignore_ascii_case("false")
487    });
488    let profiler_store = Arc::new(ProfileStore::new(50));
489    let _ = tracing_subscriber::registry()
490        .with(ProfilerLogLayer)
491        .with(tracing_subscriber::fmt::layer().with_target(true))
492        .try_init();
493
494    let state = Arc::new(AppState {
495        store,
496        index,
497        csrf: HmacCsrfTokenManager::new(csrf_secret.as_bytes()),
498        matcher: UrlMatcher::new(routes(&admin_path)?),
499        admin_token,
500        translator,
501        negotiator,
502        profiler: Arc::clone(&profiler_store),
503        notifier: NullTransport::new(),
504        theme_css,
505        theme_preload_image,
506        footer_html,
507        show_admin_link: admin_path == "/admin",
508        favicon_file,
509    });
510
511    let state_for_handler = Arc::clone(&state);
512    let mut async_kernel = AsyncHttpKernel::from_sync(move |request: &mut Request| {
513        handle(state_for_handler.as_ref(), request)
514    });
515    async_kernel.push_middleware(serenade_minify::HtmlMinifier);
516    if profiler_enabled {
517        async_kernel.push_middleware(AsyncProfilerMiddleware::new(
518            Arc::clone(&profiler_store),
519            ProfilerConfig::enabled(50),
520        ));
521    }
522
523    println!("demo-html listening on http://{bind}/");
524    println!("Admin: open {admin_path} and sign in with the configured token");
525    if profiler_enabled {
526        println!("Profiler: http://{bind}/_profiler (disable with MYFEED_PROFILER=0)");
527    }
528
529    let runtime = tokio::runtime::Runtime::new()?;
530    runtime.block_on(serenade_http_actix::listen_with_payload_limit(
531        bind,
532        async_kernel,
533        2 * MAX_IMAGE_BYTES,
534    ))?;
535    Ok(())
536}
