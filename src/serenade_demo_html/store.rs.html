11/// Moderation state for a comment.
12#[derive(Debug, Clone, Copy, PartialEq, Eq)]
13pub enum CommentStatus {
14    /// Waiting for owner approval.
15    Pending,
16    /// Visible on the public feed.
17    Approved,
18}
19
20impl CommentStatus {
21    fn parse(raw: &str) -> Self {
22        if raw == "approved" {
23            Self::Approved
24        } else {
25            Self::Pending
26        }
27    }
28}

30/// One wall post.

112    /// Opens (or creates) the SQLite database at `path` and migrates schema.
113    ///
114    /// # Panics
115    ///
116    /// Panics when the database cannot be opened or migrated (demo fail-fast).
117    #[must_use]
118    pub fn open(path: impl AsRef<Path>) -> Self {
119        let conn = Connection::open(path.as_ref()).expect("open MyFeed sqlite");
120        conn.execute_batch(
121            "
122            PRAGMA foreign_keys = ON;
123            CREATE TABLE IF NOT EXISTS posts (
124              id INTEGER PRIMARY KEY AUTOINCREMENT,
125              body TEXT NOT NULL,
126              embed_url TEXT,
127              image_data TEXT,
128              category TEXT NOT NULL,
129              likes INTEGER NOT NULL DEFAULT 0,
130              created_at TEXT NOT NULL DEFAULT (datetime('now'))
131            );
132            CREATE TABLE IF NOT EXISTS comments (
133              id INTEGER PRIMARY KEY AUTOINCREMENT,
134              post_id INTEGER NOT NULL,
135              body TEXT NOT NULL,
136              status TEXT NOT NULL CHECK (status IN ('pending', 'approved')),
137              FOREIGN KEY (post_id) REFERENCES posts(id) ON DELETE CASCADE
138            );
139            CREATE TABLE IF NOT EXISTS categories (
140              name TEXT PRIMARY KEY
141            );
142            CREATE INDEX IF NOT EXISTS comments_post_status_id
143              ON comments (post_id, status, id);
144            ",
145        )
146        .expect("migrate MyFeed sqlite");
147        ensure_created_at_column(&conn);
148        let store = Self {
149            conn: Arc::new(Mutex::new(conn)),
150        };
151        store.ensure_default_categories();
152        store
153    }

165            .query_row("SELECT COUNT(*) FROM categories", [], |row| row.get(0))

202    /// Indexed keyset window; images are never read into feed/search results.
203    pub fn feed_batch(
204        &self,
205        ceiling: u64,
206        before: Option<u64>,
207        after: Option<u64>,
208        around: Option<u64>,
209    ) -> rusqlite::Result<FeedBatch> {
210        let conn = self.conn.lock().expect("feed lock");
211        let (condition, boundary, order) = match (before, after, around) {
212            (Some(id), _, _) => ("id < ?2", id, "DESC"),
213            (_, Some(id), _) => ("id > ?2", id, "ASC"),
214            (_, _, Some(id)) => ("id <= ?2", id, "DESC"),
215            _ => ("id <= ?2", ceiling, "DESC"),
216        };
217        let sql = format!(
218            "SELECT {POST_SUMMARY} FROM posts WHERE id <= ?1 AND {condition}
219             ORDER BY id {order} LIMIT {FEED_PAGE_SIZE}"
220        );
221        let mut stmt = conn.prepare(&sql)?;
222        let mut posts = stmt
223            .query_map(params![sql_id(ceiling), sql_id(boundary)], map_summary)?
224            .collect::<rusqlite::Result<Vec<_>>>()?;
225        if after.is_some() {
226            posts.reverse();
227        }
228        let has_older = if let Some(post) = posts.last() {
229            conn.query_row(
230                "SELECT EXISTS(SELECT 1 FROM posts WHERE id < ?1)",
231                params![sql_id(post.id)],
232                |row| row.get(0),
233            )?
234        } else {
235            false
236        };
237        let has_newer = if let Some(post) = posts.first() {
238            conn.query_row(
239                "SELECT EXISTS(SELECT 1 FROM posts WHERE id > ?1 AND id <= ?2)",
240                params![sql_id(post.id), sql_id(ceiling)],
241                |row| row.get(0),
242            )?
243        } else {
244            false
245        };
246        Ok(FeedBatch {
247            posts,
248            ceiling,
249            has_older,
250            has_newer,
251        })
252    }

279    pub fn comment_batch(
280        &self,
281        post_id: u64,
282        before: Option<u64>,
283        after: Option<u64>,
284    ) -> rusqlite::Result<CommentBatch> {
285        let conn = self.conn.lock().expect("feed lock");
286        let (condition, boundary, order) = before.map_or_else(
287            || ("id > ?2", after.unwrap_or(0), "ASC"),
288            |id| ("id < ?2", id, "DESC"),
289        );
290        let sql = format!(
291            "SELECT id, post_id, body, status FROM comments
292             WHERE post_id = ?1 AND status = 'approved' AND {condition}
293             ORDER BY id {order} LIMIT {COMMENT_PAGE_SIZE}"
294        );
295        let mut stmt = conn.prepare(&sql)?;
296        let mut comments = stmt
297            .query_map(params![sql_id(post_id), sql_id(boundary)], map_comment)?
298            .collect::<rusqlite::Result<Vec<_>>>()?;
299        if before.is_some() {
300            comments.reverse();
301        }
302        debug_assert!(comments.iter().all(|c| c.status == CommentStatus::Approved));
303        let exists = |condition: &str, id: u64| -> rusqlite::Result<bool> {
304            conn.query_row(
305                &format!(
306                    "SELECT EXISTS(SELECT 1 FROM comments
307                    WHERE post_id = ?1 AND status = 'approved' AND {condition})"
308                ),
309                params![sql_id(post_id), sql_id(id)],
310                |row| row.get(0),
311            )
312        };
313        let has_older = comments
314            .first()
315            .map_or(Ok(false), |c| exists("id < ?2", c.id))?;
316        let has_newer = comments
317            .last()
318            .map_or(Ok(false), |c| exists("id > ?2", c.id))?;
319        Ok(CommentBatch {
320            comments,
321            has_older,
322            has_newer,
323        })
324    }

356    /// Adds a category (slugified, max 40 chars, unique case-insensitive).
357    pub fn add_category(&self, name: &str) -> Result<(), &'static str> {
358        let name = name.trim();
359        if name.is_empty() {
360            return Err("Category name is required.");
361        }
362        let name = slug(name);
363        if name.is_empty() {
364            return Err("Category name is required.");
365        }
366        let name: String = name.chars().take(40).collect();
367        let conn = self.conn.lock().expect("feed lock");
368        let exists: bool = conn
369            .query_row(
370                "SELECT 1 FROM categories WHERE name = ?1 COLLATE NOCASE LIMIT 1",
371                params![name],
372                |_| Ok(true),
373            )
374            .optional()
375            .ok()
376            .flatten()
377            .unwrap_or(false);
378        if exists {
379            return Err("Category already exists.");
380        }
381        let count: i64 = conn
382            .query_row("SELECT COUNT(*) FROM categories", [], |row| row.get(0))
383            .unwrap_or(0);
384        if count >= 32 {
385            return Err("Too many categories.");
386        }
387        conn.execute("INSERT INTO categories (name) VALUES (?1)", params![name])
388            .map_err(|_| "Could not add category.")?;
389        Ok(())
390    }

460    /// Adds a post; media already allowlist / size-checked by the caller.
461    pub fn add_post(&self, new: NewPost) -> Post {
462        let conn = self.conn.lock().expect("feed lock");
463        conn.execute(
464            "INSERT INTO posts (body, embed_url, image_data, category, likes, created_at)
465             VALUES (?1, ?2, ?3, ?4, 0, datetime('now'))",
466            params![new.body, new.embed_url, new.image_data, new.category],
467        )
468        .expect("insert post");
469        let id = rust_id(conn.last_insert_rowid());
470        let created_at: String = conn
471            .query_row(
472                "SELECT created_at FROM posts WHERE id = ?1",
473                params![sql_id(id)],
474                |row| row.get(0),
475            )
476            .unwrap_or_else(|_| String::new());
477        drop(conn);
478        Post {
479            id,
480            body: new.body,
481            embed_url: new.embed_url,
482            has_image: new.image_data.is_some(),
483            image_data: new.image_data,
484            category: new.category,
485            likes: 0,
486            created_at,
487        }
488    }

595fn ensure_created_at_column(conn: &Connection) {
596    let mut stmt = conn
597        .prepare("PRAGMA table_info(posts)")
598        .expect("pragma posts");
599    let names: Vec<String> = stmt
600        .query_map([], |row| row.get::<_, String>(1))
601        .expect("pragma map")
602        .filter_map(Result::ok)
603        .collect();
604    if names.iter().any(|name| name == "created_at") {
605        return;
606    }
607    conn.execute(
608        "ALTER TABLE posts ADD COLUMN created_at TEXT NOT NULL DEFAULT ''",
609        [],
610    )
611    .expect("alter created_at");
612    conn.execute(
613        "UPDATE posts SET created_at = datetime('now') WHERE created_at = ''",
614        [],
615    )
616    .expect("backfill created_at");
617}

659    #[test]
660    fn archive_windows_are_bounded_stable_and_image_free() {
661        for count in [365, 3650, 10_000] {
662            let store = FeedStore::open_memory();
663            {
664                let mut conn = store.conn.lock().expect("feed lock");
665                let tx = conn.transaction().expect("fixture transaction");
666                {
667                    let mut insert = tx
668                        .prepare(
669                            "INSERT INTO posts (body, category, image_data) VALUES
670                        ('<p>fixture</p>', 'life', 'data:image/png;base64,aGVsbG8=')",
671                        )
672                        .expect("fixture insert");
673                    for _ in 0..count {
674                        insert.execute([]).expect("seed fixture");
675                    }
676                }
677                tx.commit().expect("commit fixture");
678            }
679            let ceiling = store.latest_id().expect("latest");
680            let new = store.add_post(NewPost {
681                body: "new during reading".into(),
682                embed_url: None,
683                image_data: None,
684                category: "life".into(),
685            });
686            assert!(store.delete_post(ceiling - 25));
687            let mut before = None;
688            let mut seen = Vec::new();
689            loop {
690                let batch = store
691                    .feed_batch(ceiling, before, None, None)
692                    .expect("batch");
693                assert!(batch.posts.len() <= FEED_PAGE_SIZE);
694                assert!(
695                    batch
696                        .posts
697                        .iter()
698                        .all(|p| p.image_data.is_none() && p.has_image)
699                );
700                assert!(batch.posts.windows(2).all(|pair| pair[0].id > pair[1].id));
701                seen.extend(batch.posts.iter().map(|p| p.id));
702                if !batch.has_older {
703                    break;
704                }
705                before = batch.posts.last().map(|p| p.id);
706            }
707            assert_eq!(seen.len(), count - 1);
