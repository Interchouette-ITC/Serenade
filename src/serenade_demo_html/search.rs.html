9fn post_search_document(post: &Post) -> SearchDocument {

30fn query_param(query: Option<&str>, name: &str) -> Option<String> {
31    let query = query?;
32    for part in query.split('&') {
33        let (key, value) = part.split_once('=').unwrap_or((part, ""));
34        if key == name {
35            return Some(percent_decode(value));
36        }
37    }
38    None
39}

41fn percent_decode(raw: &str) -> String {
42    let mut out = Vec::with_capacity(raw.len());
43    let bytes = raw.as_bytes();
44    let mut i = 0;
45    while i < bytes.len() {
46        match bytes[i] {
47            b'+' => {
48                out.push(b' ');
49                i += 1;
50            }
51            b'%' if i + 2 < bytes.len() => {
52                let hi = from_hex(bytes[i + 1]);
53                let lo = from_hex(bytes[i + 2]);
54                if let (Some(h), Some(l)) = (hi, lo) {
55                    out.push((h << 4) | l);
56                    i += 3;
57                } else {
58                    out.push(b'%');
59                    i += 1;
60                }
61            }
62            b => {
63                out.push(b);
64                i += 1;
65            }
66        }
67    }
68    String::from_utf8_lossy(&out).into_owned()
69}

71const fn from_hex(byte: u8) -> Option<u8> {
72    match byte {
73        b'0'..=b'9' => Some(byte - b'0'),
74        b'a'..=b'f' => Some(byte - b'a' + 10),
75        b'A'..=b'F' => Some(byte - b'A' + 10),
76        _ => None,
77    }
78}

81fn snippet_for(post: &Post) -> String {
82    let plain = plain_text(&post.body);
83    let trimmed: String = plain.chars().take(160).collect();
84    if plain.chars().count() > 160 {
85        format!("{trimmed}…")
86    } else if trimmed.is_empty() {
87        format!("#{}", post.id)
88    } else {
89        trimmed
90    }
91}

93pub fn handle_search(state: &AppState, request: &Request) -> Response {
94    let ui = ui_for(state, request);
95    let q = query_param(request.query(), "q").unwrap_or_default();
96    let mut owned: Vec<(String, String, String)> = Vec::new();
97    if !q.trim().is_empty() {
98        let hits = state
99            .index
100            .query(&SearchQuery::new(q.clone()).with_limit(50))
101            .unwrap_or_default();
102        for hit in hits {
103            let Ok(id) = hit.id().parse::<u64>() else {
104                continue;
105            };
106            let Some(post) = state.store.get_post(id) else {
107                continue;
108            };
109            owned.push((
110                format!("/#post-{id}"),
111                snippet_for(&post),
112                post.category.clone(),
113            ));
114        }
115    }
116    let hit_views: Vec<SearchHitView<'_>> = owned
117        .iter()
118        .map(|(href, snippet, category)| SearchHitView {
119            href,
120            snippet,
121            category,
122        })
123        .collect();
124    html_response(
125        200,
126        search_page(&ui, state.matcher.collection(), &q, &hit_views),
127    )
128}
