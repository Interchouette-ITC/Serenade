36fn cookie_value(request: &Request, name: &str) -> Option<String> {
37    let header = request.headers().get("cookie")?;
38    for part in header.split(';') {
39        let part = part.trim();
40        if let Some((key, value)) = part.split_once('=')
41            && key.trim() == name
42        {
43            return Some(value.trim().to_owned());
44        }
45    }
46    None
47}

49pub fn is_admin(request: &Request, expected: &str) -> bool {
50    if request.headers().get("Authorization").is_some_and(|value| {
51        let key = value.strip_prefix("Bearer ").unwrap_or(value);
52        key == expected
53    }) {
54        return true;
55    }
56    cookie_value(request, ADMIN_COOKIE).is_some_and(|value| value == expected)
57}

59pub fn ui_for<'a>(state: &'a AppState, request: &Request) -> Ui<'a> {
60    let locale = serenade_translation::request_locale(request)
61        .cloned()
62        .unwrap_or_else(|| state.negotiator.resolve(request));
63    state.translator.set_locale(locale.clone());
64    Ui::new(&state.translator, locale)
65}

67pub fn handle_locale_set(state: &AppState, request: &Request) -> Response {
68    let code = request
69        .attributes()
70        .get::<String>("code")
71        .map_or("en", String::as_str);
72    let locale = Locale::new(code).unwrap_or_else(|_| Locale::new("en").expect("en"));
73    let matched = state
74        .negotiator
75        .allowed()
76        .iter()
77        .find(|item| item.language() == locale.language())
78        .cloned()
79        .unwrap_or_else(|| state.negotiator.default_locale().clone());
80    let cookie = format!(
81        "{LOCALE_COOKIE}={tag}; Path=/; SameSite=Lax; Max-Age=31536000",
82        tag = matched.as_str()
83    );
84    redirect_with_cookie("/", &cookie)
85}
