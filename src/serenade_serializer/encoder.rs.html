3use serde_json::Value;

7/// JSON format name used by the built-in codecs.
8pub const FORMAT_JSON: &str = "json";
9
10/// TOON v4.1 format name (LLM / agent context export). Requires feature `toon`.
11#[cfg(feature = "toon")]
12pub const FORMAT_TOON: &str = "toon";
13
14/// Encodes an intermediate [`Value`] into bytes for a format.
15pub trait Encoder: Send + Sync {
16    /// Returns `true` when this encoder handles `format`.
17    fn supports(&self, format: &str) -> bool;
18
19    /// Encodes `data` for `format`.
20    ///
21    /// # Errors
22    ///
23    /// Returns [`SerializerError::Codec`] when encoding fails, or
24    /// [`SerializerError::UnsupportedFormat`] when the format is not supported.
25    fn encode(&self, data: &Value, format: &str) -> Result<Vec<u8>, SerializerError>;
26}
27
28/// Decodes bytes into an intermediate [`Value`] for a format.
29pub trait Decoder: Send + Sync {
30    /// Returns `true` when this decoder handles `format`.
31    fn supports(&self, format: &str) -> bool;
32
33    /// Decodes `data` for `format`.
34    ///
35    /// # Errors
36    ///
37    /// Returns [`SerializerError::Codec`] when decoding fails, or
38    /// [`SerializerError::UnsupportedFormat`] when the format is not supported.
39    fn decode(&self, data: &[u8], format: &str) -> Result<Value, SerializerError>;
40}
41
42/// JSON encoder (`serde_json`).
43#[derive(Debug, Default, Clone, Copy)]
44pub struct JsonEncoder;
45
46impl Encoder for JsonEncoder {
47    fn supports(&self, format: &str) -> bool {
48        format.eq_ignore_ascii_case(FORMAT_JSON)
49    }
50
51    fn encode(&self, data: &Value, format: &str) -> Result<Vec<u8>, SerializerError> {
52        if !self.supports(format) {
53            return Err(SerializerError::UnsupportedFormat {
54                format: format.to_owned(),
55            });
56        }
57        serde_json::to_vec(data).map_err(|err| crate::error::from_serde_json(&err))
58    }
59}

61/// JSON decoder (`serde_json`).
62#[derive(Debug, Default, Clone, Copy)]
63pub struct JsonDecoder;
64
65impl Decoder for JsonDecoder {
66    fn supports(&self, format: &str) -> bool {
67        format.eq_ignore_ascii_case(FORMAT_JSON)
68    }
69
70    fn decode(&self, data: &[u8], format: &str) -> Result<Value, SerializerError> {
71        if !self.supports(format) {
72            return Err(SerializerError::UnsupportedFormat {
73                format: format.to_owned(),
74            });
75        }
76        serde_json::from_slice(data).map_err(|err| crate::error::from_serde_json(&err))
77    }
78}

101/// TOON v4.1 decoder (`reddb-io-toon`).
102#[cfg(feature = "toon")]
103#[derive(Debug, Default, Clone, Copy)]
104pub struct ToonDecoder;
105
106#[cfg(feature = "toon")]
107impl Decoder for ToonDecoder {
108    fn supports(&self, format: &str) -> bool {
109        format.eq_ignore_ascii_case(FORMAT_TOON)
110    }
111
112    fn decode(&self, data: &[u8], format: &str) -> Result<Value, SerializerError> {
113        if !self.supports(format) {
114            return Err(SerializerError::UnsupportedFormat {
115                format: format.to_owned(),
116            });
117        }
118        let text = std::str::from_utf8(data).map_err(|err| SerializerError::Codec {
119            message: err.to_string(),
120        })?;
121        let toon_value = reddb_io_toon::decode(text).map_err(|err| SerializerError::Codec {
122            message: err.to_string(),
123        })?;
124        Ok(toon_value.to_json_value())
125    }
126}

128/// Encodes an intermediate [`Value`] as a UTF-8 TOON string for LLM prompts.
129///
130/// # Errors
131///
132/// Returns [`SerializerError::Codec`] when TOON encoding fails.
133///
134/// # Examples
135///
136/// ```
137/// use serde_json::json;
138/// use serenade_serializer::encode_toon_string;
139///
140/// let text = encode_toon_string(&json!({ "sku": "HOODIE", "qty": 2 })).expect("ok");
141/// assert!(text.contains("sku"));
142/// ```
143#[cfg(feature = "toon")]
144pub fn encode_toon_string(data: &Value) -> Result<String, SerializerError> {
145    encode_toon_string_with_options(data, reddb_io_toon::EncodeOptions::default())
146}
147
148#[cfg(feature = "toon")]
149fn encode_toon_string_with_options(
150    data: &Value,
151    options: reddb_io_toon::EncodeOptions,
152) -> Result<String, SerializerError> {
153    let toon_value = reddb_io_toon::Value::from_json_value(data.clone());
154    reddb_io_toon::encode_with_options(&toon_value, options).map_err(|err| SerializerError::Codec {
155        message: err.to_string(),
156    })
157}
158
159#[cfg(feature = "toon")]
160fn encode_toon_bytes(data: &Value) -> Result<Vec<u8>, SerializerError> {
161    Ok(encode_toon_string(data)?.into_bytes())
162}

165mod encode_error_tests {
166    use super::{Encoder, FORMAT_TOON, ToonEncoder, encode_toon_string_with_options};
167    use crate::SerializerError;
168    use serde_json::json;
169
170    #[test]
171    fn encode_rejects_shallow_max_depth() {
172        let options = reddb_io_toon::EncodeOptions {
173            max_depth: 1,
174            ..Default::default()
175        };
176        let err = encode_toon_string_with_options(&json!({ "a": { "b": { "c": 1 } } }), options)
177            .expect_err("too deep");
178        assert!(matches!(err, SerializerError::Codec { .. }));
179    }
180
181    #[test]
182    fn toon_encoder_roundtrip_still_works() {
183        let bytes = ToonEncoder
184            .encode(&json!({ "ok": true }), FORMAT_TOON)
185            .expect("encode");
186        assert!(!bytes.is_empty());
187    }
