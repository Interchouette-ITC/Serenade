3use std::cmp::Ordering;
4use std::path::{Path, PathBuf};
5
6use walkdir::WalkDir;
7
8use crate::error::FinderError;
9use crate::glob::name_matches;

11/// Entry type filter.
12#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
13pub enum EntryKind {
14    /// Files and directories.
15    #[default]
16    Any,
17    /// Regular files only.
18    File,
19    /// Directories only.
20    Directory,
21}
22
23/// Symfony-shaped directory walker with filters.
24#[derive(Debug, Clone, Default)]
25pub struct Finder {
26    roots: Vec<PathBuf>,
27    kind: EntryKind,
28    names: Vec<String>,
29    min_depth: usize,
30    max_depth: Option<usize>,
31    ignore_dotfiles: bool,
32    follow_links: bool,
33}
34
35impl Finder {
36    /// Empty finder (call [`Self::in_path`] before [`Self::collect`]).
37    #[must_use]
38    pub fn new() -> Self {
39        Self::default()
40    }
41
42    /// Add a search root directory.
43    #[must_use]
44    pub fn in_path(mut self, path: impl AsRef<Path>) -> Self {
45        self.roots.push(path.as_ref().to_path_buf());
46        self
47    }
48
49    /// Restrict to regular files.
50    #[must_use]
51    pub const fn files(mut self) -> Self {
52        self.kind = EntryKind::File;
53        self
54    }
55
56    /// Restrict to directories.
57    #[must_use]
58    pub const fn directories(mut self) -> Self {
59        self.kind = EntryKind::Directory;
60        self
61    }
62
63    /// Basename must match at least one `*` / `?` pattern.
64    #[must_use]
65    pub fn name(mut self, pattern: impl Into<String>) -> Self {
66        self.names.push(pattern.into());
67        self
68    }
69
70    /// Minimum depth relative to each root (0 = root itself).
71    #[must_use]
72    pub const fn min_depth(mut self, depth: usize) -> Self {
73        self.min_depth = depth;
74        self
75    }
76
77    /// Maximum depth relative to each root (inclusive).
78    #[must_use]
79    pub const fn max_depth(mut self, depth: usize) -> Self {
80        self.max_depth = Some(depth);
81        self
82    }
83
84    /// Skip entries whose basename starts with `.`.
85    #[must_use]
86    pub const fn ignore_dotfiles(mut self) -> Self {
87        self.ignore_dotfiles = true;
88        self
89    }
90
91    /// Follow symbolic links while walking.
92    #[must_use]
93    pub const fn follow_links(mut self) -> Self {
94        self.follow_links = true;
95        self
96    }
97
98    /// Collect matching paths sorted by path string.
99    ///
100    /// # Errors
101    ///
102    /// Empty roots, invalid root, or walk I/O failure.
103    pub fn collect(self) -> Result<Vec<PathBuf>, FinderError> {
104        if self.roots.is_empty() {
105            return Err(FinderError::EmptyRoots);
106        }
107        let mut out = Vec::new();
108        for root in &self.roots {
109            if !root.is_dir() {
110                return Err(FinderError::InvalidRoot { path: root.clone() });
111            }
112            let mut walker = WalkDir::new(root).min_depth(self.min_depth);
113            if let Some(max) = self.max_depth {
114                walker = walker.max_depth(max);
115            }
116            if self.follow_links {
117                walker = walker.follow_links(true);
118            }
119            for entry in walker {
120                let entry = entry.map_err(|err| FinderError::Walk {
121                    path: err.path().map_or_else(|| root.clone(), Path::to_path_buf),
122                    source: std::io::Error::other(err.to_string()),
123                })?;
124                let path = entry.path();
125                let file_type = entry.file_type();
126                if self.ignore_dotfiles
127                    && path
128                        .file_name()
129                        .and_then(|s| s.to_str())
130                        .is_some_and(|name| name.starts_with('.') && path != root.as_path())
131                {
132                    continue;
133                }
134                let ok_kind = match self.kind {
135                    EntryKind::Any => true,
136                    EntryKind::File => file_type.is_file(),
137                    EntryKind::Directory => file_type.is_dir(),
138                };
139                if !ok_kind {
140                    continue;
141                }
142                if !self.names.is_empty() {
143                    let Some(name) = path.file_name().and_then(|s| s.to_str()) else {
144                        continue;
145                    };
146                    if !self.names.iter().any(|pat| name_matches(pat, name)) {
147                        continue;
148                    }
149                }
150                out.push(path.to_path_buf());
151            }
152        }
153        out.sort_by(|a, b| cmp_paths(a, b));
154        Ok(out)
155    }
156}
157
158fn cmp_paths(a: &Path, b: &Path) -> Ordering {
159    a.as_os_str().cmp(b.as_os_str())
