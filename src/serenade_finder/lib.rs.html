1//! Directory walking with name glob and type filters (Symfony Finder shaped).
2//!
3//! - [`Finder`] - `in_path`, `files` / `directories`, `name`, depth, `ignore_dotfiles`
4//! - [`Finder::collect`] - sorted matching paths
5//!
6//! # Examples
7//!
8//! ```
9//! use serenade_finder::Finder;
10//! use std::fs;
11//!
12//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
13//! let dir = tempfile_dir()?;
14//! fs::write(dir.join("a.rs"), b"")?;
15//! fs::write(dir.join("b.txt"), b"")?;
16//! let found = Finder::new().in_path(&dir).files().name("*.rs").collect()?;
17//! assert_eq!(found.len(), 1);
18//! # Ok(())
19//! # }
20//! # fn tempfile_dir() -> std::io::Result<std::path::PathBuf> {
21//! #   let dir = std::env::temp_dir().join(format!("serenade-finder-doc-{}", std::process::id()));
22//! #   fs::create_dir_all(&dir)?;
23//! #   Ok(dir)
24//! # }
25//! ```
